```ruby
module V1
  module Account
    class AdminBrokeragesController < ApplicationController
      include MainDomainOnly
      include CognitoAuth

      before_action :authenticate_cognito_user!

      def skip_brokerage_membership?
        true
      end

      # GET /v1/account/admin_brokerages
      def index
        admin_memberships = BrokerageMembership
          .without_deleted
          .includes(:brokerage)
          .where(user_id: Current.user.id, role: %w[admin owner])   # <-- line to change

        deduped_memberships = admin_memberships
          .each_with_object({}) { |m, h| h[m.brokerage_id] = m if m.brokerage && (h[m.brokerage_id].nil? || m.role == "owner") }
          .values

        brokerage_ids = deduped_memberships.filter_map(&:brokerage_id)
        return render json: {admin_brokerages: []} if brokerage_ids.empty?

        all_memberships = BrokerageMembership
          .without_deleted
          .joins(:user)
          .includes(:user)
          .where(brokerage_id: brokerage_ids, users: {deleted_at: nil})

        memberships_by_brokerage = all_memberships.group_by(&:brokerage_id)
        member_roles = %w[agent staff owner admin]   # already includes agent — no change needed

        render json: {
          admin_brokerages: deduped_memberships.map { |m|
            group = memberships_by_brokerage[m.brokerage_id] || []
            {
              id: m.brokerage.id,
              name: m.brokerage.name,
              role: m.role,
              client_count: group.count { |r| r.role == "client" },
              member_count: group.count { |r| member_roles.include?(r.role) },
              owners: group.select { |r| r.role == "owner" }.map { |r| {first_name: r.user.first_name, last_name: r.user.last_name} }
            }
          }
        }
      end
    end
  end
end
```

```ruby
module AuthHelper
  # Mints a real HS256 JWT via TokenService.issue. Sets Current.brokerage BEFORE
  # calling issue so the role is embedded correctly. Uses Host header (subdomain path).
  def auth_headers_for(user, brokerage)
    Current.brokerage = brokerage
    Current.user = user
    token = TokenService.issue(user)
    { "Authorization" => "Bearer #{token}", "Host" => brokerage.brokerage_domains.first.domain }
  ensure
    Current.reset
  end
end
```
This controller requires `MainDomainOnly` (Origin header must resolve to `APP_BASE_DOMAIN`, or `localhost`/`127.0.0.1` in test/dev) — `auth_headers_for` (Host-based, subdomain path) is NOT suitable here. Use the `main_domain_headers_for`-style pattern instead (see `backend/test/integration/brokerage_auth_main_domain_test.rb`, full relevant section already read):
```ruby
def main_domain_headers_for(user)
  Current.user = user
  token = TokenService.issue(user)
  { "Authorization" => "Bearer #{token}", "Origin" => "https://#{ENV.fetch('APP_BASE_DOMAIN', 'realtybase.ca')}" }
ensure
  Current.reset
end
```
Note: `TokenService.issue` embeds `role` from `BrokerageMembership.find_by(user_id:, brokerage_id: Current.brokerage&.id)` — for this controller `Current.brokerage` is irrelevant to authorization (it `skip_brokerage_membership?`s), but a role claim is still safe to omit/ignore since `CognitoAuth` only checks `Current.user` for this controller. Do not set `Current.brokerage` before issuing the token in these tests — leave it nil, matching the `main_domain_headers_for` pattern.
