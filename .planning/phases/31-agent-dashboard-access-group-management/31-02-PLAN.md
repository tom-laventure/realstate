```tsx
const RequireRole = ({ roles, redirectTo = '/', children }: RequireRoleProps) => {
    const { role, isUserLoaded } = useRole()
    // Bypasses entirely on main domain — has zero effect on the main-domain path (Plan 31-01 covers that path).
    if (isMainDomain()) return <>{children}</>
    if (!isUserLoaded) return null
    if (!role || !roles.includes(role)) return <Navigate to={redirectTo} replace />
    return <>{children}</>
}
```
