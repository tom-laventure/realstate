```typescript
import { signOut } from 'aws-amplify/auth'
import { useNavigate } from 'react-router-dom'
import { Check, ChevronDown } from 'lucide-react'
import { Button } from '@/components/ui/button'
import {
    DropdownMenu,
    DropdownMenuContent,
    DropdownMenuItem,
    DropdownMenuTrigger,
} from '@/components/ui/dropdown-menu'
import { clearToken } from '@/lib/auth'
import { useAppSelector, useAppDispatch } from '@/store/hooks/useDispatch'
import { setActiveBrokerage } from '@/store/reducers/activeBrokerage'
import type { AdminBrokerage } from '@/assets/types/adminBrokerageType'

const PortalNavBar = () => {
    const activeBrokerage = useAppSelector((s) => s.activeBrokerage.activeBrokerage)
    const adminBrokerages = useAppSelector((s) => s.activeBrokerage.adminBrokerages)
    const dispatch = useAppDispatch()
    const navigate = useNavigate()

    // D-12: Null guard — rendered only once activeBrokerage is set in Redux by PortalShell.
    if (!activeBrokerage) return null

    const handleSignOut = async () => {
        await signOut({ global: true }).catch(() => null)
        clearToken()
        window.location.href = '/auth/login'
    }

    // D-14: dispatch first, then navigate — both must execute unconditionally.
    const handleSwitch = (b: AdminBrokerage) => {
        dispatch(setActiveBrokerage(b))
        navigate(`/dashboard/${b.id}`)
    }

    return (
        <header className="border-b border-sidebar-border bg-sidebar">
            <div className="flex h-16 items-center justify-between gap-3 px-4">
                {/* Left: brokerage switcher (D-11, D-13) */}
                <div>
                    <DropdownMenu>
                        <DropdownMenuTrigger asChild>
                            <Button
                                variant="ghost"
                                className="flex items-center gap-1.5 px-2 text-sm font-semibold text-sidebar-foreground hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
                            >
                                {activeBrokerage.name}
                                <ChevronDown className="h-4 w-4 text-muted-foreground" />
                            </Button>
                        </DropdownMenuTrigger>
                        <DropdownMenuContent align="start" className="w-52">
                            {adminBrokerages.map((b) => (
                                <DropdownMenuItem
                                    key={b.id}
                                    onClick={() => handleSwitch(b)}
                                    className="flex items-center justify-between"
                                >
                                    <span>{b.name}</span>
                                    {b.id === activeBrokerage.id && (
                                        <Check className="h-4 w-4 text-primary" />
                                    )}
                                </DropdownMenuItem>
                            ))}
                        </DropdownMenuContent>
                    </DropdownMenu>
                </div>

                {/* Right: role badge + sign-out (D-11) */}
                <div className="flex items-center gap-2">
                    <span className="inline-flex items-center rounded-full bg-muted px-2 py-0.5 text-xs text-muted-foreground capitalize">
                        {activeBrokerage.role}
                    </span>
                    <Button
                        variant="ghost"
                        size="sm"
                        onClick={handleSignOut}
                        className="text-sidebar-foreground hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
                    >
                        Sign Out
                    </Button>
                </div>
            </div>
        </header>
    )
}

export default PortalNavBar
```

Key constraints:
- Do NOT add `signingOut` state — the UI-SPEC and PATTERNS.md explicitly say not to (PortalNavBar sign-out is a small ghost button, not a full-page button like NoAccess).
- Do NOT use shadcn `<Card>` or `<Badge>` — role badge is an inline `<span>` per established portal pattern.
- `handleSwitch` must call dispatch AND navigate — never make navigate conditional.
- `DropdownMenuContent align="start"` (not "end") — opens below the left edge of the trigger.
