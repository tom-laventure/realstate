```ruby
def auth_headers_for(user, brokerage)
  Current.brokerage = brokerage
  Current.user = user
  token = TokenService.issue(user)

ensure
  Current.reset
end
```

```ruby
require "test_helper"

class TenantIsolationTest < ActionDispatch::IntegrationTest
  setup do
    @brokerage_a = create(:brokerage)
    @domain_a    = create(:brokerage_domain, brokerage: @brokerage_a, domain: "a.example.com")
    @brokerage_b = create(:brokerage)
    @domain_b    = create(:brokerage_domain, brokerage: @brokerage_b, domain: "b.example.com")
    @user_a      = create(:user)
    create(:brokerage_membership, :owner, user: @user_a, brokerage: @brokerage_a)
  end

  # A valid JWT for Brokerage A combined with Brokerage B's Host header must be rejected.
  # BrokerageAuth resolves tenant from Host → finds Brokerage B.
  # CognitoAuth then checks membership of @user_a in Brokerage B → none → 403 "no-access".
  # Alternatively, if BrokerageAuth raises RecordNotFound, Rails returns 404.
  # Either outcome proves cross-tenant isolation holds.
  test "Brokerage A JWT against Brokerage B host is rejected with 403 or 404" do
    headers = auth_headers_for(@user_a, @brokerage_a).merge("Host" => @domain_b.domain)
    get "/v1/admin/users", headers: headers
    assert_includes [403, 404], response.status
  end

  # Smoke test: same user, same brokerage, correct Host → must be accepted (200 or any non-4xx).
  # Verifies setup is correct and not broken by test isolation issues.
  test "Brokerage A JWT against Brokerage A host is accepted (owner role)" do
    get "/v1/admin/users", headers: auth_headers_for(@user_a, @brokerage_a)
    assert_response :ok
  end
end
```

```ruby
require "test_helper"

class AdminAuthorizationTest < ActionDispatch::IntegrationTest
  setup do
    @brokerage = create(:brokerage)
    create(:brokerage_domain, brokerage: @brokerage, domain: "test.example.com")

    @owner  = create(:user)
    @admin  = create(:user)
    @agent  = create(:user)
    @client = create(:user)

    create(:brokerage_membership, :owner,  user: @owner,  brokerage: @brokerage)
    create(:brokerage_membership, :admin,  user: @admin,  brokerage: @brokerage)
    create(:brokerage_membership, :agent,  user: @agent,  brokerage: @brokerage)
    create(:brokerage_membership, :client, user: @client, brokerage: @brokerage)
  end

  # --- Rejection cases ---

  test "agent role receives 403 on GET /v1/admin/users" do
    get "/v1/admin/users", headers: auth_headers_for(@agent, @brokerage)
    assert_response :forbidden
    body = JSON.parse(response.body)
    assert_equal "forbidden", body.dig("error", "code")
  end

  test "client role receives 403 on GET /v1/admin/users" do
    get "/v1/admin/users", headers: auth_headers_for(@client, @brokerage)
    assert_response :forbidden
    body = JSON.parse(response.body)
    assert_equal "forbidden", body.dig("error", "code")
  end

  # --- Acceptance cases ---

  test "owner role receives 200 on GET /v1/admin/users" do
    get "/v1/admin/users", headers: auth_headers_for(@owner, @brokerage)
    assert_response :ok
  end

  test "admin role receives 200 on GET /v1/admin/users" do
    get "/v1/admin/users", headers: auth_headers_for(@admin, @brokerage)
    assert_response :ok
  end

  # --- Unauthenticated case ---

  test "request with no JWT receives 401 on GET /v1/admin/users" do
    get "/v1/admin/users", headers: { "Host" => "test.example.com" }
    assert_response :unauthorized
  end
end
```
