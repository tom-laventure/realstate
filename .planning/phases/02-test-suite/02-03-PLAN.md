```ruby
class FilterSetsTest < ActionDispatch::IntegrationTest
  setup do

    # Attach two estates to @group under the active filter set
    estate_1 = create(:estate)
    estate_2 = create(:estate)
    @ge_1 = create(:group_estate, group: @group, estate: estate_1, group_filter_set_id: @active_filter_set.id)
    @ge_2 = create(:group_estate, group: @group, estate: estate_2, group_filter_set_id: @active_filter_set.id)

    # Attach one estate to @other_group (must NOT be deleted when @group filter set changes)
    estate_3 = create(:estate)
    @other_ge = create(:group_estate, group: @other_group, estate: estate_3, group_filter_set_id: nil)
  end

  test "creating a new filter set destroys previous active filter set estates for this group" do
    assert_equal 2, GroupEstate.where(group: @group, group_filter_set_id: @active_filter_set.id).count
```

```ruby
require "test_helper"

class GroupEstatesQueryTest < ActionDispatch::IntegrationTest
  setup do
    @brokerage = create(:brokerage)
    create(:brokerage_domain, brokerage: @brokerage, domain: "test.example.com")
    @user = create(:user)
    create(:brokerage_membership, :owner, user: @user, brokerage: @brokerage)
    @group = create(:group, brokerage: @brokerage)

    # Create estates with known prices
    @cheap_estate    = create(:estate)
    @expensive_estate = create(:estate)
    @mid_estate      = create(:estate)

    create(:listing_detail, estate: @cheap_estate,     price_numeric: 100_000.0)
    create(:listing_detail, estate: @expensive_estate, price_numeric: 900_000.0)
    create(:listing_detail, estate: @mid_estate,       price_numeric: 500_000.0)

    @cheap_ge    = create(:group_estate, group: @group, estate: @cheap_estate)
    @expensive_ge = create(:group_estate, group: @group, estate: @expensive_estate)
    @mid_ge      = create(:group_estate, group: @group, estate: @mid_estate)
  end

  # --- Price range filter ---

  test "price range filter returns only estates within range" do
    Current.brokerage = @brokerage
    Current.user = @user
    results = GroupEstatesQuery.new(@group, @user, { min_price: "200000", max_price: "600000" }).call
    estate_ids = results.map(&:estate_id)
    assert_includes estate_ids, @mid_ge.estate_id
    assert_not_includes estate_ids, @cheap_ge.estate_id
    assert_not_includes estate_ids, @expensive_ge.estate_id
  ensure
    Current.reset
  end

  # --- Price sort ---

  test "sort_by price-asc returns cheapest estate first" do
    Current.brokerage = @brokerage
    Current.user = @user
    results = GroupEstatesQuery.new(@group, @user, { sort_by: "price-asc" }).call
    prices = results.map { |ge| ge.estate.listing_detail.price_numeric }
    assert_equal prices, prices.sort
  ensure
    Current.reset
  end

  test "sort_by price-desc returns most expensive estate first" do
    Current.brokerage = @brokerage
    Current.user = @user
    results = GroupEstatesQuery.new(@group, @user, { sort_by: "price-desc" }).call
    prices = results.map { |ge| ge.estate.listing_detail.price_numeric }
    assert_equal prices, prices.sort.reverse
  ensure
    Current.reset
  end

  # --- Rating sort ---

  test "sort_by rating-desc returns highest-rated estate first" do
    # Add ratings — @expensive_ge gets a high rating, @cheap_ge gets no rating
    estate_rating = EstateRating.create!(
      group_estate_id: @expensive_ge.id,
      user_id: @user.id,
      rating: 5
    )

    Current.brokerage = @brokerage
    Current.user = @user
    results = GroupEstatesQuery.new(@group, @user, { sort_by: "rating-desc" }).call
    first_ge = results.first
    assert_equal @expensive_ge.id, first_ge.id, "Estate with highest rating must appear first"
  ensure
    Current.reset
  end
end
```

Note on `EstateRating.create!`: check `estate_ratings` schema columns (group_estate_id, user_id, rating). If there is a unique index on [group_estate_id, user_id], only one rating per user per group_estate can exist — the test creates exactly one, which is valid.
