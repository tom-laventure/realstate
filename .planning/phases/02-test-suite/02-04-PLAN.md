Create the invitee with a JWT from a brokerage they ARE a member of for auth, but the invitation is for the TARGET brokerage. require "test_helper"

```ruby
class InvitationsTest < ActionDispatch::IntegrationTest
  setup do
    @brokerage = create(:brokerage)
    create(:brokerage_domain, brokerage: @brokerage, domain: "test.example.com")

    @inviter = create(:user)
    create(:brokerage_membership, :owner, user: @inviter, brokerage: @brokerage)

    @invitee = create(:user)
    # invitee is a member of the brokerage so auth passes (skip_brokerage_membership? = true for accept_token)
    create(:brokerage_membership, :agent, user: @invitee, brokerage: @brokerage)

    @invitation = create(:invitation,
      brokerage: @brokerage,
      inviter: @inviter,
      email: @invitee.email,
      role: "agent",
      expires_at: 7.days.from_now
    )
  end

  # --- accept_token ---

  # This test proves the Phase 1 Current.user fix holds.
  # Before the fix, this action crashed with NoMethodError: undefined method `current_user`.
  test "accept_token with valid invite_token returns 200" do
    post "/v1/dashboard/invitations/accept_token",
      headers: auth_headers_for(@invitee, @brokerage),
      params: { invite_token: @invitation.token },
      as: :json

    assert_response :ok
    body = JSON.parse(response.body)
    assert body.key?("message"), "Response must include 'message' key"
  end

  test "accept_token with invalid invite_token returns 404 or 422" do
    post "/v1/dashboard/invitations/accept_token",
      headers: auth_headers_for(@invitee, @brokerage),
      params: { invite_token: "invalid-token-does-not-exist" },
      as: :json

    assert_includes [404, 422], response.status
  end

  # --- validate_token (public — no auth) ---

  test "validate_token with valid token returns 200 with invitation details" do
    post "/v1/dashboard/invitations/validate_token",
      headers: { "Host" => "test.example.com" },
      params: { invite_token: @invitation.token },
      as: :json

    assert_response :ok
    body = JSON.parse(response.body)
    assert body.key?("invitation"), "Response must include 'invitation' key"
  end

  test "validate_token with invalid token returns 404" do
    post "/v1/dashboard/invitations/validate_token",
      headers: { "Host" => "test.example.com" },
      params: { invite_token: "not-a-real-token" },
      as: :json

    assert_response :not_found
  end
end
```

| T-02-04-01 | Spoofing | validate_token public access | accept | validate_token intentionally has no auth (email links call it before sign-in); token itself is a 64-char random hex — brute force infeasible; no sensitive data returned beyond brokerage name and invitee email |
| T-02-04-02 | Elevation of Privilege | accept_token skip_brokerage_membership | accept | skip_brokerage_membership? returns true for accept_token only; user still requires a valid Cognito JWT (authenticate_cognito_user! is NOT skipped); the action creates a membership, not bypasses one |
| T-02-04-03 | Tampering | Invitation token replay after acceptance | accept | Invitation model has accepted_at; once accepted, a second accept_token call for the same token would find the invitation already accepted — controller handles this gracefully |
