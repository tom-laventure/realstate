```ruby
require_relative "../config/environment"

Shoulda::Matchers.configure do |config|
  config.integrate do |with|
    with.test_framework :minitest
    with.library :rails
  end
end

class ActiveSupport::TestCase
  include FactoryBot::Syntax::Methods

  teardown do
    Current.reset
  end
end

module AuthHelper
  # Mints a real HS256 JWT via TokenService.issue.
  # Sets Current.brokerage BEFORE calling issue so the role is embedded correctly.
  # Returns headers hash with Authorization + Host suitable for integration test requests.
  def auth_headers_for(user, brokerage)
    Current.brokerage = brokerage
    Current.user = user
    token = TokenService.issue(user)
    {
      "Authorization" => "Bearer #{token}",
      "Host" => brokerage.brokerage_domains.first.domain
    }
  ensure
    Current.reset
  end
end

class ActionDispatch::IntegrationTest
  include AuthHelper
end
```
