```typescript
export const isAuthenticated = (): boolean => {
  const token = getToken()  // localStorage.getItem('auth_token')
  if (!token) return false
  const payload = decodePayload(token)  // base64-decodes the middle segment
  return typeof payload?.exp === 'number' && payload.exp > Date.now() / 1000
}
// To mock authenticated: localStorage.setItem('auth_token', 'header.' + btoa(JSON.stringify({ exp: Math.floor(Date.now()/1000) + 3600 })) + '.sig')
```

```typescript
const RequireRole = ({ roles, redirectTo = '/', children }: RequireRoleProps) => {
  const { role, isUserLoaded } = useRole()
  if (!isUserLoaded) return null
  if (!role || !roles.includes(role)) {
    return <Navigate to={redirectTo} replace />
  }
  return <>{children}</>
}
```

```typescript
import { screen } from '@testing-library/react'
import { describe, it, expect, beforeEach } from 'vitest'
import { Route, Routes } from 'react-router-dom'
import ProtectedRoute from './ProtectedRoute'
import { renderWithProviders } from '@/test/renderWithProviders'

// Helper: create a valid JWT with exp in the future
function makeValidToken(): string {
  const payload = btoa(JSON.stringify({ exp: Math.floor(Date.now() / 1000) + 3600 }))
  return `header.${payload}.sig`
}

describe('ProtectedRoute', () => {
  beforeEach(() => {
    localStorage.clear() // ensure clean state even though setup.ts does this afterEach
  })

  it('redirects unauthenticated users to /auth/login', () => {
    // localStorage is empty — isAuthenticated() returns false
    renderWithProviders(
      <Routes>
        <Route element={<ProtectedRoute />}>
          <Route path="/" element={<div>Protected Content</div>} />
        </Route>
        <Route path="/auth/login" element={<div>Login Page</div>} />
      </Routes>,
      { route: '/' }
    )
    expect(screen.getByText('Login Page')).toBeInTheDocument()
    expect(screen.queryByText('Protected Content')).not.toBeInTheDocument()
  })

  it('renders protected content for authenticated users', () => {
    localStorage.setItem('auth_token', makeValidToken())
    renderWithProviders(
      <Routes>
        <Route element={<ProtectedRoute />}>
          <Route path="/" element={<div>Protected Content</div>} />
        </Route>
        <Route path="/auth/login" element={<div>Login Page</div>} />
      </Routes>,
      { route: '/' }
    )
    expect(screen.getByText('Protected Content')).toBeInTheDocument()
    expect(screen.queryByText('Login Page')).not.toBeInTheDocument()
  })
})
```

```typescript
import { screen } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { describe, it, expect, vi, beforeEach } from 'vitest'
import GoogleSignInButton from './GoogleSignInButton'
import { renderWithProviders } from '@/test/renderWithProviders'

// Mock aws-amplify sub-path modules (v6 pattern — mock sub-paths, not root package)
const mockSignInWithRedirect = vi.fn()
const mockFetchAuthSession = vi.fn().mockResolvedValue({
  tokens: { idToken: { toString: () => 'mock-id-token' } }
})
let mockHubCallback: ((event: { payload: { event: string } }) => void) | null = null

vi.mock('aws-amplify/auth', () => ({
  signInWithRedirect: mockSignInWithRedirect,
  fetchAuthSession: mockFetchAuthSession,
}))

vi.mock('aws-amplify/utils', () => ({
  Hub: {
    listen: vi.fn().mockImplementation((_channel, callback) => {
      mockHubCallback = callback
      // Return an unlisten function
      return () => { mockHubCallback = null }
    }),
  },
}))

describe('GoogleSignInButton', () => {
  beforeEach(() => {
    vi.clearAllMocks()
    mockHubCallback = null
  })

  it('calls signInWithRedirect with Google provider when clicked', async () => {
    const user = userEvent.setup()
    renderWithProviders(<GoogleSignInButton />)

    await user.click(screen.getByRole('button', { name: /sign in with google/i }))

    expect(mockSignInWithRedirect).toHaveBeenCalledOnce()
    expect(mockSignInWithRedirect).toHaveBeenCalledWith({ provider: 'Google' })
  })

  it('calls fetchAuthSession when Hub fires signInWithRedirect event', async () => {
    renderWithProviders(<GoogleSignInButton />)

    // Hub.listen should have been called during useEffect
    expect(mockHubCallback).not.toBeNull()

    // Simulate Hub firing the signInWithRedirect event
    mockHubCallback!({ payload: { event: 'signInWithRedirect' } })

    // fetchAuthSession should be called asynchronously
    await vi.waitFor(() => {
      expect(mockFetchAuthSession).toHaveBeenCalledOnce()
    })
  })
})
```
