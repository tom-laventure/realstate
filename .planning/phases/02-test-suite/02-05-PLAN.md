**frontend/src/test/renderWithProviders.tsx:**
Read `frontend/src/store/reducers/index.tsx` first to get the exact reducer import paths. Then:
```typescript
import React from 'react'
import { render } from '@testing-library/react'
import { QueryClient, QueryClientProvider } from '@tanstack/react-query'
import { Provider } from 'react-redux'
import { MemoryRouter } from 'react-router-dom'
import { configureStore } from '@reduxjs/toolkit'
import accountReducer from '@/store/reducers/account'
import createBrokerageWizardReducer from '@/store/reducers/createBrokerageWizard'
import tempBrokerageReducer from '@/store/reducers/tempBrokerage'

interface RenderOptions {
  route?: string
  preloadedState?: Record<string, unknown>
}

export function renderWithProviders(
  ui: React.ReactElement,
  { route = '/', preloadedState = {} }: RenderOptions = {}
) {
  // Fresh QueryClient per test — prevents cache leakage between tests
  const queryClient = new QueryClient({
    defaultOptions: {
      queries: { retry: false },
      mutations: { retry: false },
    },
  })

  // Fresh Redux store per test — prevents state leakage between tests
  const store = configureStore({
    reducer: {
      account: accountReducer,
      createBrokerageWizard: createBrokerageWizardReducer,
      tempBrokerage: tempBrokerageReducer,
    },
    preloadedState,
  })

  return render(
    <QueryClientProvider client={queryClient}>
      <Provider store={store}>
        <MemoryRouter initialEntries={[route]}>
          {ui}
        </MemoryRouter>
      </Provider>
    </QueryClientProvider>
  )
}
```

Note: If `frontend/src/store/reducers/index.tsx` uses different import paths or reducer names, use those exact paths. The three reducer imports above (`account`, `createBrokerageWizard`, `tempBrokerage`) are derived from the existing `src/store/reducers/index.tsx` — verify before writing.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| MSW network interception | MSW intercepts fetch/XHR in jsdom; with `onUnhandledRequest: 'warn'`, unhandled requests warn rather than throw — prevents test failures from unexpected network calls during setup |
| renderWithProviders Redux store | preloadedState allows tests to seed any Redux state; this is test-only infrastructure, never ships to production |
