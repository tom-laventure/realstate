```ruby
require "test_helper"

class BrokerageTest < ActiveSupport::TestCase
  setup do
    @brokerage = create(:brokerage)
    @other_brokerage = create(:brokerage)
  end

  test "phasing_out_count_for returns 0 when no phasing-out domains exist" do
    create(:brokerage_domain, brokerage: @brokerage, status: "active", domain_type: "subdomain", domain: "active.example.com")
    assert_equal 0, @brokerage.phasing_out_count_for(:subdomain)
    assert_equal 0, @brokerage.phasing_out_count_for(:custom)
  end

  test "phasing_out_count_for counts only the requested type and only phasing_out status" do
    create(:brokerage_domain, brokerage: @brokerage, status: "phasing_out", domain_type: "subdomain", domain: "p1.example.com", is_primary: false, phase_out_at: 10.days.from_now)
    create(:brokerage_domain, brokerage: @brokerage, status: "phasing_out", domain_type: "subdomain", domain: "p2.example.com", is_primary: false, phase_out_at: 20.days.from_now)
    create(:brokerage_domain, brokerage: @brokerage, status: "active",      domain_type: "subdomain", domain: "a1.example.com", is_primary: true)
    create(:brokerage_domain, brokerage: @brokerage, status: "phasing_out", domain_type: "custom",    domain: "p1.custom.com", is_primary: false, phase_out_at: 5.days.from_now)

    assert_equal 2, @brokerage.phasing_out_count_for(:subdomain)
    assert_equal 1, @brokerage.phasing_out_count_for(:custom)
  end

  test "phasing_out_count_for is scoped to the brokerage" do
    create(:brokerage_domain, brokerage: @brokerage,       status: "phasing_out", domain_type: "subdomain", domain: "p.mine.com",  is_primary: false, phase_out_at: 10.days.from_now)
    create(:brokerage_domain, brokerage: @other_brokerage, status: "phasing_out", domain_type: "subdomain", domain: "p.other.com", is_primary: false, phase_out_at: 10.days.from_now)
    assert_equal 1, @brokerage.phasing_out_count_for(:subdomain)
    assert_equal 1, @other_brokerage.phasing_out_count_for(:subdomain)
  end
end
```
