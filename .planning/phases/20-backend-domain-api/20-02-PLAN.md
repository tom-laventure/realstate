```ruby
module V1
  module Admin
    class DomainsController < V1::Admin::BaseController
      # require_admin! is the default from BaseController — applies to index (D-10).
      # Mutations require owner role (D-11).
      before_action :require_owner!, only: %i[subdomain custom destroy set_primary]

      # GET /v1/admin/domains
      # DMOD-06: returns all active and phasing-out domains for the current brokerage.
      def index
        domains = Current.brokerage.brokerage_domains.routable.order(:created_at)
        render json: serialize_collection(domains), status: :ok
      end

      # PATCH /v1/admin/domains/subdomain
      # DMOD-07: phase out current active subdomain, create new active subdomain.
      def subdomain
        slug = params[:slug].to_s.strip.downcase
        return render json: { error: "slug is required" }, status: :bad_request if slug.blank?

        if (blocked = slot_block_response(:subdomain))
          return blocked
        end

        full_host = "#{slug}.#{base_domain}"
        old_primary = Current.brokerage.brokerage_domains.status_active.domain_type_subdomain.first
        inherit_primary = old_primary&.is_primary? || false

        new_domain = nil
        ApplicationRecord.transaction do
          old_primary&.phase_out!
          # If we phased-out the old primary, primary flag moves to the new record.
          # If no old primary existed, new record stays non-primary.
          new_domain = Current.brokerage.brokerage_domains.create!(
            domain: full_host,
            domain_type: :subdomain,
            status: :active,
            is_primary: inherit_primary,
            verified: true,
            verified_at: Time.current,
            verification_token: nil
          )
        end

        render json: mutation_payload(new_domain, :subdomain), status: :ok
      rescue ActiveRecord::RecordInvalid => e
        render json: { error: "Failed to update subdomain", details: e.record.errors.full_messages }, status: :unprocessable_entity
      end

      # POST /v1/admin/domains/custom
      # DMOD-08: add custom domain (premium only); phase out existing active custom if any.
      def custom
        unless Current.brokerage.plan == "premium"
          return render json: { error: "Custom domains require a premium plan", code: "plan_required" }, status: :forbidden
        end

        domain_param = params[:domain].to_s.strip.downcase
        return render json: { error: "domain is required" }, status: :bad_request if domain_param.blank?

        if (blocked = slot_block_response(:custom))
          return blocked
        end

        old_active_custom = Current.brokerage.brokerage_domains.status_active.domain_type_custom.first
        new_domain = nil
        ApplicationRecord.transaction do
          old_active_custom&.phase_out!
          # Assumption A1 (20-RESEARCH.md): create with verified: true since DNS verification is deferred to v1.6.
          # status: :active requires verified: true to satisfy status_requires_verified validation.
          new_domain = Current.brokerage.brokerage_domains.create!(
            domain: domain_param,
            domain_type: :custom,
            status: :active,
            is_primary: false,
            verified: true,
            verified_at: Time.current,
            verification_token: nil
          )
        end

        render json: mutation_payload(new_domain, :custom), status: :ok
      rescue ActiveRecord::RecordInvalid => e
        render json: { error: "Failed to add custom domain", details: e.record.errors.full_messages }, status: :unprocessable_entity
      end

      # DELETE /v1/admin/domains/:id
      # DMOD-09: active custom -> phase out (with slot check).
      # DMOD-14: phasing_out (any type) -> immediate decommission (frees slot).
      # Subdomain (active): not directly deletable — use PATCH /subdomain instead (Open Question 2).
      def destroy
        domain = Current.brokerage.brokerage_domains.find_by(id: params[:id])
        return render json: { error: "Domain not found" }, status: :not_found unless domain

        if domain.status_decommissioned?
          return render json: { error: "Domain cannot be deleted in its current state" }, status: :unprocessable_entity
        end

        if domain.status_phasing_out?
          # DMOD-14 — immediate decommission, frees a slot.
          domain.decommission!
          return render json: mutation_payload(domain, domain.domain_type.to_sym), status: :ok
        end

        # From here, status is :active.
        if domain.domain_type_subdomain?
          return render json: {
            error: "Cannot remove subdomain directly; use PATCH /subdomain to change it",
            code: "subdomain_not_deletable"
          }, status: :unprocessable_entity
        end

        # DMOD-09: active custom -> phase out (with slot block check).
        if (blocked = slot_block_response(:custom))
          return blocked
        end

        domain.phase_out!
        render json: mutation_payload(domain, :custom), status: :ok
      rescue ActiveRecord::RecordInvalid => e
        render json: { error: "Failed to delete domain", details: e.record.errors.full_messages }, status: :unprocessable_entity
      end

      # PATCH /v1/admin/domains/:id/set_primary
      # DMOD-10: designate an ACTIVE domain as primary; reject phasing_out / decommissioned with 422.
      def set_primary
        domain = Current.brokerage.brokerage_domains.find_by(id: params[:id])
        return render json: { error: "Domain not found" }, status: :not_found unless domain

        unless domain.status_active?
          return render json: {
            error: "Only active domains can be set as primary",
            code: "invalid_status"
          }, status: :unprocessable_entity
        end

        ApplicationRecord.transaction do
          Current.brokerage.brokerage_domains
            .where(is_primary: true)
            .update_all(is_primary: false)
          domain.update!(is_primary: true)
        end

        render json: mutation_payload(domain.reload, domain.domain_type.to_sym), status: :ok
      rescue ActiveRecord::RecordInvalid => e
        render json: { error: "Failed to set primary", details: e.record.errors.full_messages }, status: :unprocessable_entity
      end

      private

      def base_domain
        ENV.fetch("APP_BASE_DOMAIN", "realstate.ca")
      end

      # Pre-action guard for DMOD-12.
      # Returns a render-able response (controller should `return` it) if blocked, nil otherwise.
      def slot_block_response(domain_type)
        return nil if Current.brokerage.phasing_out_count_for(domain_type) < 2

        earliest = Current.brokerage.brokerage_domains
          .status_phasing_out
          .where(domain_type: domain_type)
          .order(:phase_out_at)
          .first
          .phase_out_at

        render json: {
          error: "Phase-out slot limit reached",
          earliest_slot_available_at: earliest
        }, status: :unprocessable_entity
      end

      # Post-action warning for DMOD-11.
      # Returns a warning hash if phasing-out count for the type equals 2, else nil.
      def phase_out_warning_for(domain_type)
        count = Current.brokerage.phasing_out_count_for(domain_type)
        return nil unless count == 2

        earliest = Current.brokerage.brokerage_domains
          .status_phasing_out
          .where(domain_type: domain_type)
          .order(:phase_out_at)
          .first
          .phase_out_at

        { earliest_slot_available_at: earliest }
      end

      def serialize_collection(collection)
        {
          domains: ActiveModelSerializers::SerializableResource.new(
            collection,
            each_serializer: BrokerageDomainSerializer,
            adapter: :attributes
          ).as_json
        }
      end

      # D-04: mutation responses bundle { domain, domains, phase_out_warning? }.
      def mutation_payload(updated_domain, domain_type)
        all_domains = Current.brokerage.brokerage_domains.routable.order(:created_at)
        payload = {
          domain: ActiveModelSerializers::SerializableResource.new(
            updated_domain,
            serializer: BrokerageDomainSerializer,
            adapter: :attributes
          ).as_json,
          domains: ActiveModelSerializers::SerializableResource.new(
            all_domains,
            each_serializer: BrokerageDomainSerializer,
            adapter: :attributes
          ).as_json
        }
        if (warning = phase_out_warning_for(domain_type))
          payload[:phase_out_warning] = warning
        end
        payload
      end
    end
  end
end
```
