import { ArrowRight } from 'lucide-react'

import type { EstateCoordinate } from '@/assets/types/estateCoordinateType'

interface GroupMapClusterPopupProps {
    estateIds: string[]
    coordinates: EstateCoordinate[]
    groupId: string
}

const GroupMapClusterPopup = ({ estateIds, coordinates, groupId }: GroupMapClusterPopupProps) => {
    const coordById = new Map(coordinates.map(c => [c.id, c]))
    const rows = estateIds
        .map(id => coordById.get(id))

            <p className="mb-2 text-sm font-semibold">

            <div className="divide-y divide-border">

                    <div

                        className="flex items-center justify-between gap-4 py-1 text-sm"

                        <span className="shrink-0 font-semibold">

1. **Create `frontend/src/components/dashboard/general/GroupMapPanel.tsx`** with this exact content:

```tsx
import { useCallback, useEffect, useRef, useState } from 'react'
import { useParams } from 'react-router-dom'
import { Loader2, MapPin } from 'lucide-react'
import {
    APIProvider,
    Map,
    AdvancedMarker,
    InfoWindow,
    useMap,
} from '@vis.gl/react-google-maps'
import {
    MarkerClusterer,
    defaultOnClusterClickHandler,
} from '@googlemaps/markerclusterer'
import type { Marker, Cluster } from '@googlemaps/markerclusterer'
import type { EstateCoordinate } from '@/assets/types/estateCoordinateType'
import GroupMapPinPopup from './GroupMapPinPopup'
import GroupMapClusterPopup from './GroupMapClusterPopup'

const MAP_DEFAULT_CENTER = { lat: 49.28, lng: -123.12 }
const MAP_DEFAULT_ZOOM = 11
const MAX_ZOOM_FOR_CLUSTER_POPUP = 19

interface GroupMapPanelProps {
    coordinates: EstateCoordinate[]
    isLoading: boolean
    isError: boolean
}

interface ClusterPopupState {
    position: { lat: number; lng: number }
    estateIds: string[]
}

const PanelShell = ({ children, dashed = false }: { children: React.ReactNode; dashed?: boolean }) => (
    <div
        className={`flex h-96 w-full flex-col items-center justify-center gap-3 rounded-xl border text-center ${
            dashed ? 'border-dashed' : 'border-border'
        }`}
    >
        {children}
    </div>
)

const MapContent = ({
    coordinates,
    groupId,
}: {
    coordinates: EstateCoordinate[]
    groupId: string
}) => {
    const map = useMap()
    const clustererRef = useRef<MarkerClusterer | null>(null)
    const markersRef = useRef<Map<string, Marker>>(new globalThis.Map())
    const reverseMarkersRef = useRef<WeakMap<Marker, string>>(new WeakMap())
    const [markers, setMarkers] = useState<Record<string, Marker>>({})
    const [selectedPin, setSelectedPin] = useState<EstateCoordinate | null>(null)
    const [clusterPopup, setClusterPopup] = useState<ClusterPopupState | null>(null)
    const hasFitBoundsRef = useRef(false)

    // Cluster click handler — show popup at max zoom, default zoom-in otherwise
    const handleClusterClick = useCallback(
        (event: google.maps.MapMouseEvent, cluster: Cluster, mapInstance: google.maps.Map) => {
            const currentZoom = mapInstance.getZoom() ?? 0
            if (currentZoom >= MAX_ZOOM_FOR_CLUSTER_POPUP) {
                const ids: string[] = []
                ;(cluster.markers ?? []).forEach(m => {
                    const id = reverseMarkersRef.current.get(m as Marker)
                    if (id) ids.push(id)
                })
                setSelectedPin(null)
                setClusterPopup({
                    position: { lat: cluster.position.lat(), lng: cluster.position.lng() },
                    estateIds: ids,
                })
            } else {
                defaultOnClusterClickHandler(event, cluster, mapInstance)
            }
        },
        []
    )

    // Initialize the clusterer once the map instance is ready
    useEffect(() => {
        if (!map) return
        clustererRef.current = new MarkerClusterer({ map, onClusterClick: handleClusterClick })
        return () => {
            clustererRef.current?.clearMarkers()
            clustererRef.current = null
        }
    }, [map, handleClusterClick])

    // Sync marker set into the clusterer whenever the React markers state changes
    useEffect(() => {
        const clusterer = clustererRef.current
        if (!clusterer) return
        clusterer.clearMarkers()
        clusterer.addMarkers(Object.values(markers))
    }, [markers])

    // Fit bounds once on initial data load only (avoids jarring re-centering on every filter)
    useEffect(() => {
        if (!map || hasFitBoundsRef.current || coordinates.length === 0) return
        const bounds = new google.maps.LatLngBounds()
        coordinates.forEach(c => bounds.extend({ lat: c.latitude, lng: c.longitude }))
        map.fitBounds(bounds, 40)
        hasFitBoundsRef.current = true
    }, [map, coordinates])

    const setMarkerRef = useCallback((marker: Marker | null, id: string) => {
        setMarkers(prev => {
            if (marker && prev[id]) return prev
            if (!marker && !prev[id]) return prev
            const next = { ...prev }
            if (marker) {
                next[id] = marker
                markersRef.current.set(id, marker)
                reverseMarkersRef.current.set(marker, id)
            } else {
                const existing = prev[id]
                if (existing) reverseMarkersRef.current.delete(existing)
                markersRef.current.delete(id)
                delete next[id]
            }
            return next
        })
    }, [])

    return (
        <>
            {coordinates.map(coord => (
                <AdvancedMarker
                    key={coord.id}
                    position={{ lat: coord.latitude, lng: coord.longitude }}
                    ref={(marker: Marker | null) => setMarkerRef(marker, coord.id)}
                    onClick={() => {
                        setClusterPopup(null)
                        setSelectedPin(coord)
                    }}
                />
            ))}

            {selectedPin && (
                <InfoWindow
                    position={{ lat: selectedPin.latitude, lng: selectedPin.longitude }}
                    onCloseClick={() => setSelectedPin(null)}
                >
                    <GroupMapPinPopup coordinate={selectedPin} groupId={groupId} />
                </InfoWindow>
            )}

            {clusterPopup && (
                <InfoWindow
                    position={clusterPopup.position}
                    onCloseClick={() => setClusterPopup(null)}
                >
                    <GroupMapClusterPopup
                        estateIds={clusterPopup.estateIds}
                        coordinates={coordinates}
                        groupId={groupId}
                    />
                </InfoWindow>
            )}
        </>
    )
}

const GroupMapPanel = ({ coordinates, isLoading, isError }: GroupMapPanelProps) => {
    const apiKey = import.meta.env.VITE_GOOGLE_MAPS_API_KEY || ''
    const mapId = import.meta.env.VITE_GOOGLE_MAPS_MAP_ID || ''
    const { groupId } = useParams<{ groupId?: string }>()

    if (!apiKey) {
        return (
            <PanelShell dashed>
                <p className="text-sm text-muted-foreground">Google Maps API key not configured</p>
            </PanelShell>
        )
    }

    if (isLoading && coordinates.length === 0) {
        return (
            <PanelShell>
                <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
            </PanelShell>
        )
    }

    if (isError) {
        return (
            <PanelShell dashed>
                <MapPin className="h-8 w-8 text-muted-foreground" />
                <div>
                    <p className="font-medium">Couldn&apos;t load map</p>
                    <p className="text-sm text-muted-foreground">
                        Location data failed to load. Try refreshing the page.
                    </p>
                </div>
            </PanelShell>
        )
    }

    if (coordinates.length === 0) {
        return (
            <PanelShell dashed>
                <MapPin className="h-8 w-8 text-muted-foreground" />
                <div>
                    <p className="font-medium">No listings with location data</p>
                    <p className="text-sm text-muted-foreground">
                        Location data isn&apos;t available for the listings in this group yet.
                    </p>
                </div>
            </PanelShell>
        )
    }

    return (
        <APIProvider apiKey={apiKey}>
            <div className="h-96 w-full overflow-hidden rounded-xl border border-border">
                <Map
                    mapId={mapId || 'group-dashboard-map'}
                    defaultCenter={MAP_DEFAULT_CENTER}
                    defaultZoom={MAP_DEFAULT_ZOOM}
                    style={{ width: '100%', height: '100%' }}
                    gestureHandling="cooperative"
                    disableDefaultUI={false}
                >
                    <MapContent coordinates={coordinates} groupId={groupId ?? ''} />
                </Map>
            </div>
        </APIProvider>
    )
}

export default GroupMapPanel
```

CRITICAL implementation notes:
- `useMap()` is called inside `MapContent` which is a child of `<Map>` — required by the library. If you put `MapContent` as a sibling of `<Map>`, `useMap()` returns null and clustering breaks.
- `mapId` is REQUIRED for `AdvancedMarker`. The fallback `'group-dashboard-map'` is a non-functional placeholder; without a real Map ID from Google Cloud Console, markers will silently not render. The `.env.example` documents the setup; runtime degradation is acceptable.
- The clusterer uses our custom `handleClusterClick` — at zoom < 19 it delegates to `defaultOnClusterClickHandler`; at zoom >= 19 it sets `clusterPopup` state instead of zooming further.
- `WeakMap<Marker, string>` (`reverseMarkersRef`) is the reverse lookup that maps an `AdvancedMarkerElement` (what `cluster.markers` contains) back to the estate id. Updated in the same callback as `setMarkers`.
- `keepPreviousData: true` on the hook (Plan 02) means existing pins remain during a filter refetch. The panel only shows the loading spinner if `isLoading && coordinates.length === 0` (initial load), not on subsequent refetches.
- `fitBounds` runs once via `hasFitBoundsRef` — UI-SPEC requires no re-centering on filter changes.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Google Maps JS SDK (HTTPS) | API key is embedded in the bundle (intended — `VITE_` env vars are public by design) |
| InfoWindow content → DOM | Address/price strings from `EstateCoordinate` interpolated as React text (auto-escaped by React) |
| Link href → Router | `groupId` from useParams + `coordinate.id` from API response interpolated into pathname |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-09-04-01 | Tampering | API key extracted from bundle | mitigate | Documented in Plan 03's user_setup — user must restrict the API key to allowed HTTP referrers in Google Cloud Console. Without this, abuse is possible but limited to billing impact, not data exposure. |
| T-09-04-02 | Information Disclosure | InfoWindow content | accept | All popup data (address, price, hero URL) is already returned by `/estates` index to the same authenticated user. No new data exposure. React auto-escapes text interpolation, mitigating XSS via address strings. |
| T-09-04-03 | Tampering | Link href injection | mitigate | `coordinate.id` is server-controlled (returned by the backend `#coordinates` action which `.to_s`'s an integer PK — no user input in the path). `groupId` comes from `useParams` (already URL-validated by react-router). No risk of crafted hrefs. |
| T-09-04-04 | Spoofing | Map ID misconfiguration | accept | If Map ID is missing, AdvancedMarker silently renders nothing. Documented in user_setup; degradation is visible (no pins) and is fixed by adding the env var. |
| T-09-04-05 | Denial of Service | Cluster reverse-lookup with 1000+ markers | accept | Each marker takes ~constant time to add to clusterer; reverse-lookup is O(1) via WeakMap. At 500 markers (realistic max), memory is well under 1MB. No mitigation required for v1.2. |
| T-09-04-06 | Repudiation | User clicks "View details" — navigation | accept | Client-side routing; no user action audit required for this v1 phase. Server-side endpoints already log access. |
