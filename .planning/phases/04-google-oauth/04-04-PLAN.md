      fill="#EA4335"
    />
  </svg>
)

const GoogleSignInButton = () => {
  const [isLoading, setIsLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const handleClick = async () => {
    setError(null)
    setIsLoading(true)

    try {
      // ~500ms spinner for visual feedback (per D-12) before browser navigates away
      await new Promise((resolve) => setTimeout(resolve, 500))

      // Encode brokerage domain in customState (per D-13)
      // Callback page reads this from Amplify's Hub event to know which tenant to redirect to
      const customState = btoa(JSON.stringify({ brokerage: window.location.hostname }))

      await signInWithRedirect({
        provider: 'Google',
        customState,
      })
      // Browser navigates to Google — component unmounts, no further state management needed
    } catch (err) {
      // signInWithRedirect throws if Amplify config is invalid or OAuth is misconfigured
      setIsLoading(false)
      setError('Sign-in with Google failed. Please try again or use email and password.')
    }
  }

  return (
    <div className="w-full">
      <Button
        type="button"
        variant="outline"
        className="w-full min-h-[44px] flex items-center justify-center gap-2"
        onClick={handleClick}
        disabled={isLoading}
        aria-label="Continue with Google"
        aria-busy={isLoading}
      >
        {isLoading ? (
          <span
            className="h-4 w-4 animate-spin rounded-full border-2 border-primary/30 border-t-primary"
            aria-hidden="true"
          />
        ) : (
          <GoogleIcon />
        )}
        Continue with Google
      </Button>

      {error && (
        <div className="mt-2 text-sm text-destructive" role="alert">
          {error}
        </div>
      )}
    </div>
  )
}

export default GoogleSignInButton

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-04-01 | Tampering | customState encoding in GoogleSignInButton | accept | `btoa(JSON.stringify({ brokerage: window.location.hostname }))` uses the actual current hostname — cannot be tampered with client-side since the user is already on that domain. The backend sets `auth_provider = 'google'` only in the verified OAuth path (Plan 01). No security-relevant action is triggered by the badge. |
| T-04-04-03 | Information Disclosure | amplify.ts redirectSignIn public URL | accept | The Cognito callback URL `https://auth.realtybase.ca/oauth/callback` is a public, registered URL — not a secret. It must be registered in the Cognito user pool OAuth settings. Exposure in the frontend bundle is expected and safe. |
