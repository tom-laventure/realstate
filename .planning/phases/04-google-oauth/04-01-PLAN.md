```ruby
module V1
  class SessionsController < ApplicationController
    include CognitoAuth
    include BrokerageAuth

    before_action :authenticate_cognito_user!, only: [:destroy]

    def skip_brokerage_membership?
      true
    end

    # POST /v1/sessions
    def create
      if params[:email].blank? || params[:password].blank?
        render json: {error: "Email and password are required"}, status: :unprocessable_entity
        return
      end

      claims = CognitoService.initiate_auth(email: params[:email], password: params[:password])

      user = User.find_or_initialize_by(cognito_sub: claims[:sub])
      user.email = claims[:email] if user.email.blank?

      if user.first_name.blank?
        full_name = claims[:name].presence || user.email.split("@").first
        parts = full_name.split(/\s+/, 2)
        user.first_name = claims[:first_name].presence || parts[0] || "User"
        user.last_name = claims[:last_name].presence || parts[1] || "Unknown"
      end

      user.name = claims[:name] if user.name.blank?
      user.save!

      render json: {token: TokenService.issue(user)}, status: :ok
    rescue CognitoService::AuthError => e
      render json: {error: e.message}, status: :unauthorized
    rescue ActiveRecord::RecordInvalid => e
      render json: {error: "Could not create user account: #{e.message}"}, status: :unprocessable_entity
    end

    # DELETE /v1/sessions
    def destroy
      CognitoService.admin_user_global_sign_out(Current.user.cognito_sub)
    rescue => e
      Rails.logger.warn("Sign-out error: #{e.class} - #{e.message}")
    ensure
      head :no_content
    end
  end
end
```

```ruby
    # POST /v1/sessions
    def create
      if params[:id_token].present?
        # OAuth path (per D-05) — Google sign-in via intermediary callback page
        # CRITICAL: Use JWT.decode(token, nil, false) — NOT TokenService.verify.
        # TokenService.verify uses HS256 (app secret); Cognito idTokens are RS256.
        # Calling TokenService.verify raises JWT::DecodeError immediately.
        # This matches the exact pattern in CognitoService#initiate_auth.
        claims, _header = JWT.decode(params[:id_token], nil, false)

        # String-keyed claims (JWT.decode returns strings, not symbols)
        email_verified = claims["email_verified"]

        unless email_verified
          render json: {error: "Google account email is not verified"}, status: :unprocessable_entity
          return
        end

        user = User.find_or_initialize_by(cognito_sub: claims["sub"])
        user.email = claims["email"] if user.email.blank?

        if user.first_name.blank?
          full_name = claims["name"].presence || user.email.split("@").first
          parts = full_name.split(/\s+/, 2)
          user.first_name = claims["given_name"].presence || parts[0] || "User"
          user.last_name = claims["family_name"].presence || parts[1] || "Unknown"
        end

        user.name = claims["name"] if user.name.blank?
        user.auth_provider = "google"
        user.save!

        render json: {token: TokenService.issue(user)}, status: :ok
      elsif params[:email].present? && params[:password].present?
        # Existing email/password path — unchanged
        claims = CognitoService.initiate_auth(email: params[:email], password: params[:password])

        user = User.find_or_initialize_by(cognito_sub: claims[:sub])
        user.email = claims[:email] if user.email.blank?

        if user.first_name.blank?
          full_name = claims[:name].presence || user.email.split("@").first
          parts = full_name.split(/\s+/, 2)
          user.first_name = claims[:first_name].presence || parts[0] || "User"
          user.last_name = claims[:last_name].presence || parts[1] || "Unknown"
        end

        user.name = claims[:name] if user.name.blank?
        user.save!

        render json: {token: TokenService.issue(user)}, status: :ok
      else
        render json: {error: "Email and password are required"}, status: :unprocessable_entity
      end
    rescue JWT::DecodeError => e
      render json: {error: "Invalid token: #{e.message}"}, status: :unauthorized
    rescue CognitoService::AuthError => e
      render json: {error: e.message}, status: :unauthorized
    rescue ActiveRecord::RecordInvalid => e
      render json: {error: "Could not create user account: #{e.message}"}, status: :unprocessable_entity
    end
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| auth.realtybase.ca → POST /v1/sessions | idToken arrives from a static page on a separate origin; not same-origin as any brokerage tenant |
| Client JWT.decode | Cognito idToken decoded without RS256 signature verification (project-established pattern) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01-01 | Spoofing | SessionsController id_token branch | mitigate | ASVS V2: verify `email_verified: true` before `find_or_initialize_by` — reject with 422 if false; prevents account takeover via unverified Google email |
| T-04-01-02 | Tampering | JWT.decode(token, nil, false) | accept | Token comes from Amplify-completed OAuth flow; Cognito has already authenticated the user; RS256 verification would add HTTP round-trip with no practical security gain given the trust chain. Project precedent (CognitoService#initiate_auth) uses same pattern. |
| T-04-01-03 | Spoofing | CORS origins proc | mitigate | ASVS V5: auth.realtybase.ca is compared as exact full-origin string `source == "https://auth.realtybase.ca"` — not a substring match or host-only check; prevents `http://auth.realtybase.ca` or `https://evil-auth.realtybase.ca` from passing |
| T-04-01-04 | Elevation of Privilege | User.find_or_initialize_by(cognito_sub:) on soft-deleted user | accept | acts_as_paranoid default scope excludes soft-deleted; a new record is created instead of restoring deleted one. Same behavior as existing email/password path in the original code — consistent, accepted limitation documented in RESEARCH.md |
| T-04-01-05 | Information Disclosure | auth_provider in UserSerializer | accept | auth_provider is a non-sensitive enum string ('google' or null); no PII; already scoped to authenticated profile endpoint |
