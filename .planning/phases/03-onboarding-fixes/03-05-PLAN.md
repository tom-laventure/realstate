const [bannerVisible, setBannerVisible] = useState(false)

    if (bannerHydrated) return
    const flag = profileQuery.data?.showWelcomeBanner

        setBannerVisible(flag)
        setBannerHydrated(true)

const dismissBanner = useCallback(() => {
    setBannerVisible(false)
    dismissMutation.mutate()
}, [dismissMutation])

const brokerageName = brokerageQuery.data?.brokerage?.name ?? ''
const groupNames = (groupsQuery.data?.groups ?? []).map((g: { name: string }) => g.name)
```
