import { useMutation, useQueryClient } from '@tanstack/react-query'
import dismissWelcome from '@/assets/api/account/dismissWelcome'
import { AxiosErrorType } from '@/assets/types/axiosErrorType'
import { accountProfileKeys } from '@/store/hooks/queryKeys'

const useDismissWelcome = () => {
    const queryClient = useQueryClient()

    return useMutation({
        mutationFn: dismissWelcome,
        onSuccess: async () => {
            await queryClient.invalidateQueries({ queryKey: accountProfileKeys.all })
        },
        onError: (err: AxiosErrorType) => {
            console.error('Failed to dismiss welcome banner:', err)
        },
    })
}

export default useDismissWelcome
```

import { useQuery } from '@tanstack/react-query'
import fetchBrokerageContact from '@/assets/api/dashboard/brokerage/fetchBrokerageContact'
import { AxiosErrorType } from '@/assets/types/axiosErrorType'
import { dashboardBrokerageKeys } from '@/store/hooks/queryKeys'

const useFetchDashboardBrokerage = () => {
    const { isLoading, data, isError, error, refetch } = useQuery({
        queryKey: dashboardBrokerageKeys.all,
        queryFn: fetchBrokerageContact,
        retry: false,
        onError: (err: AxiosErrorType) => {
            console.error('Failed to fetch dashboard brokerage:', err)
        },
    })

    return { isLoading, data, isError, error, refetch }
}

export default useFetchDashboardBrokerage
```

import React from 'react'

    groupNames,
    onDismiss,

                    onClick={onDismiss}
                    aria-label="Dismiss welcome banner"

                    <X className="h-4 w-4" />
                </Button>
            </div>
        </div>
    )
}
