```typescript
import { useCallback, useMemo } from 'react'
import { useNavigate, useParams } from 'react-router-dom'
import useFetchSelectedGroup from '@/store/hooks/Dashboard/groups/useFetchSelectedGroup'
import usePermission from '@/store/hooks/Auth/usePermission'
import useRole from '@/store/hooks/Auth/useRole'
import { isGroupsOnlyRole } from '@/assets/rbac/permissions'

const useSideNavContext = () => {
    const navigate = useNavigate()
    const { groupId } = useParams<{ groupId?: string }>()
    const canAccessAdmin = usePermission('access_admin_panel')
    const { role } = useRole()
    const isGroupsOnlyAdmin = isGroupsOnlyRole(role)

    const hasGroupContext = Boolean(groupId)
    const { data: selectedGroupData, isLoading: isLoadingSelectedGroup } = useFetchSelectedGroup(groupId, { enabled: hasGroupContext })
    const selectedGroup = selectedGroupData?.group ?? null
    const goToHome = useCallback(() => navigate('/'), [navigate])
    const navigateTo = useCallback((path: string) => navigate(path), [navigate])

    return useMemo(() => ({
        state: { groupId, hasGroupContext, selectedGroup, isLoadingSelectedGroup, canAccessAdmin, isGroupsOnlyAdmin },
        actions: { goToHome, navigateTo },
    }), [groupId, hasGroupContext, selectedGroup, isLoadingSelectedGroup, canAccessAdmin, isGroupsOnlyAdmin, goToHome, navigateTo])
}
export default useSideNavContext
```
`role` is used in this file for exactly one purpose (`isGroupsOnlyAdmin`) — grep confirms no other usage, so after deleting `isGroupsOnlyAdmin` the `useRole` import and `const { role } = useRole()` line both become removable too.
