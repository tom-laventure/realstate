import { hasPermission, type Permission } from '@/assets/rbac/permissions'
import useRole from './useRole'
const usePermission = (permission: Permission): boolean => {
    const { role, isUserLoaded } = useRole()
    if (!isUserLoaded || !role) return false
    return hasPermission(role, permission)
}
export default usePermission

    const { mutate: deleteInvitation } = useDeleteBrokerageInvitation()
    const { mutate: deleteUser } = useDeleteUserFromBrokerage()

    return (
        <AdminBoard>
            <div className={classes.usersAdminBoard + ' w-full p-6'}>
                <UsersListView
                    users={state.users}
                    groups={state.groups}
                    paginationMeta={state.usersMeta?.pagination}
                    totalUsers={state.usersMeta?.totalUsers || 0}
                    totalAdmins={state.usersMeta?.totalAdminUsers || 0}
                    onSearchUsers={actions.onSearchUsers}
                    onGroupFilterChange={actions.onGroupFilterChange}
                    onAgentFilterChange={actions.onAgentFilterChange}
                    onPageChange={actions.setUsersPage}
                    onDeleteUser={handleDeleteUser}
                />
                <InvitationsView
                    invitations={state.invitations}
                    groups={state.groups}
                    paginationMeta={state.invitationsMeta}
                    onDeleteInvite={handleDeleteInvite}
                    onInviteFilterChange={actions.onInviteFilterChange}
                    onSearchInvite={actions.onSearchInvite}
                    onPageChange={actions.setInvitationsPage}
                />
            </div>
        </AdminBoard>
    )
}
