```ruby
class DomainMailer < ApplicationMailer
  # DHLT-03, DHLT-04: Health check failure notification to all brokerage owners.
  # domain.is_primary? is checked in the template to add subdomain-fallback language.
  def health_failure(domain)
    @domain = domain
    @brokerage = domain.brokerage
    @dashboard_url = "https://#{primary_domain(@brokerage)}/admin/domains"
    mail(to: owner_emails(@brokerage), subject: "Domain health check failed: #{domain.domain}")
  end

  # DPRD-05: Plan downgrade notification — custom domains are now frozen.
  def downgrade_notice(brokerage)
    @brokerage = brokerage
    @dashboard_url = "https://#{primary_domain(@brokerage)}/admin/domains"
    mail(to: owner_emails(@brokerage), subject: "Your custom domains have been paused")
  end

  # DADD-04: Unverified pending domain is approaching the 7-day removal deadline.
  def unverified_warning(domain)
    @domain = domain
    @brokerage = domain.brokerage
    @dashboard_url = "https://#{primary_domain(@brokerage)}/admin/domains"
    mail(to: owner_emails(@brokerage), subject: "Action required: domain #{domain.domain} expires in 2 days")
  end

  private

  # D-13: Resolve owner-role recipients at send time.
  # CRITICAL: .without_deleted required — BrokerageMembership uses acts_as_paranoid.
  def owner_emails(brokerage)
    BrokerageMembership.without_deleted
                       .where(brokerage: brokerage, role: "owner")
                       .includes(:user)
                       .map { |m| m.user.email }
  end

  # Copied verbatim from InvitationMailer — shared pattern for primary domain resolution.
  def primary_domain(brokerage)
    brokerage.brokerage_domains
      .where(is_primary: true)
      .where.not(domain: "localhost")
      .first&.domain || brokerage.brokerage_domains.first&.domain
  end
end
```

```ruby
require "test_helper"

class DomainMailerTest < ActionMailer::TestCase
  setup do
    @brokerage = create(:brokerage, name: "Acme Realty", plan: "premium")
    @primary_subdomain = create(
      :brokerage_domain,
      brokerage: @brokerage,
      domain: "acme.example.com",
      domain_type: "subdomain",
      status: "active",
      is_primary: true,
      verified: true
    )
    @custom_domain = create(
      :brokerage_domain,
      brokerage: @brokerage,
      domain: "acme.com",
      domain_type: "custom",
      status: "phasing_out",
      is_primary: false,
      verified: true,
      phase_out_at: 30.days.from_now
    )
    @owner = create(:user)
    @admin = create(:user)
    create(:brokerage_membership, :owner, user: @owner, brokerage: @brokerage)
    create(:brokerage_membership, :admin, user: @admin, brokerage: @brokerage)
  end

  def full_body(mail)
    mail.parts.map { |p| p.body.to_s }.join
  end

  # --- health_failure ---

  test "health_failure sends to owner email only (not admin)" do
    mail = DomainMailer.health_failure(@custom_domain)
    assert_includes mail.to, @owner.email
    refute_includes mail.to, @admin.email
  end

  test "health_failure subject contains domain name" do
    mail = DomainMailer.health_failure(@custom_domain)
    assert_includes mail.subject, @custom_domain.domain
  end

  test "health_failure body includes domain name and dashboard link" do
    mail = DomainMailer.health_failure(@custom_domain)
    body = full_body(mail)
    assert_includes body, @custom_domain.domain
    assert_includes body, "/admin/domains"
  end

  test "health_failure body includes decommission date when phase_out_at is set" do
    mail = DomainMailer.health_failure(@custom_domain)
    body = full_body(mail)
    assert_includes body, @custom_domain.phase_out_at.strftime("%B %-d, %Y")
  end

  test "health_failure body calls out primary domain fallback when domain is_primary? true" do
    # Make the custom domain primary
    @custom_domain.update!(is_primary: true, status: "phasing_out")
    mail = DomainMailer.health_failure(@custom_domain)
    body = full_body(mail)
    # DHLT-04: must explicitly call out subdomain fallback
    assert body.include?("primary") || body.include?("fallback"),
      "health_failure email must mention primary/fallback when domain was primary (DHLT-04)"
```
