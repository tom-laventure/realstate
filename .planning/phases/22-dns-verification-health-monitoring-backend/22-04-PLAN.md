```ruby
def custom
  unless Current.brokerage.plan == "premium"
    return render json: {error: "Custom domains require a premium plan", code: "plan_required"}, status: :forbidden
  end

  domain_param = params[:domain].to_s.strip.downcase
  return render json: {error: "domain is required"}, status: :bad_request if domain_param.blank?

  new_domain = Current.brokerage.brokerage_domains.create!(
    domain: domain_param,
    domain_type: :custom,
    status: :pending,
    is_primary: false,
    verified: false
  )

  render json: mutation_payload(new_domain, :custom), status: :ok
rescue ActiveRecord::RecordInvalid => e
  render json: {error: "Failed to add custom domain", details: e.record.errors.full_messages}, status: :unprocessable_entity
end
```
