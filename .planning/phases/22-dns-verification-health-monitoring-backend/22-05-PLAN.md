require "resolv"

class DomainVerificationJob < ApplicationJob
  queue_as :background
  retry_on StandardError, wait: :polynomially_longer, attempts: 5

  # DADD-04: Unverified pending domains are removed after this many days.
  VERIFICATION_EXPIRY_DAYS = 7
  # DADD-04: Warning email is sent when pending domain is this many days old.
  WARNING_EMAIL_DAYS = 5

  def perform
    # Step 1: Process pending domains (verify CNAME, handle expiry)
    BrokerageDomain.without_deleted
                   .domain_type_custom
                   .status_pending
                   .in_batches do |batch|
      batch.each { |domain| process_pending(domain) }
    end

    # Step 2: Process phasing-out domains (rescue opportunity — DLCX-01)
    BrokerageDomain.without_deleted
                   .domain_type_custom
                   .status_phasing_out
                   .where("phase_out_at > ?", Time.current)
                   .in_batches do |batch|
      batch.each { |domain| process_phasing_out(domain) }
    end
  end

  private

  def process_pending(domain)
    age_days = (Time.current - domain.created_at) / 1.day

    # DADD-04: Auto-remove after VERIFICATION_EXPIRY_DAYS
    if age_days >= VERIFICATION_EXPIRY_DAYS
      domain.decommission!
      return
    end

    # DADD-04: Warning email at day 5 (idempotent — may send multiple times in the 5-7 day window)
    if age_days >= WARNING_EMAIL_DAYS
      DomainMailer.unverified_warning(domain).deliver_now!
      return
    end

    # Verify CNAME
    if cname_resolves_to_subdomain?(domain)
      promote_to_active!(domain)
    end
    domain.update_column(:last_checked_at, Time.current)
  end

  # DLCX-01: Rescue phasing-out domain if CNAME resolves before decommission date.
  def process_phasing_out(domain)
    if cname_resolves_to_subdomain?(domain)
      ApplicationRecord.transaction do
        locked = BrokerageDomain.lock.find(domain.id)
        # Guard: may have been decommissioned or transitioned by another worker
        return unless locked.status_phasing_out?
        locked.update!(
          status: :active,
          consecutive_failures: 0,
          last_checked_at: Time.current,
          phase_out_at: nil
        )
      end
    end
  end

  # DVER-05: Pessimistic lock prevents duplicate promotion by concurrent job runs.
  def promote_to_active!(domain)
    ApplicationRecord.transaction do
      locked = BrokerageDomain.lock.find(domain.id)
      return if locked.status_active?  # idempotency guard

      locked.update!(
        status: :active,
        verified: true,
        verified_at: Time.current,
        consecutive_failures: 0,
        last_checked_at: Time.current
      )
      # Auto-set primary if the brokerage has no primary domain yet
      if locked.brokerage.brokerage_domains.where(is_primary: true).none?
        locked.update!(is_primary: true)
      end
    end
  end

  # D-01: Use getresources (plural) — returns [] on miss, does not raise.
  # D-02: Target subdomain is looked up live from the brokerage's active subdomain.
  # D-03: Any failure (wrong target, DNS error, timeout) returns false.
  def cname_resolves_to_subdomain?(domain)
    target = domain.brokerage.brokerage_domains
                             .domain_type_subdomain
                             .status_active
                             .first
                             &.domain
    return false if target.nil?

    Resolv::DNS.open do |dns|
      records = dns.getresources(domain.domain, Resolv::DNS::Resource::IN::CNAME)
      records.any? { |r| r.name.to_s.chomp(".") == target.chomp(".") }
    end
  rescue Resolv::ResolvError, Resolv::ResolvTimeout
    false
  end
end

require "resolv"

class DomainHealthJob < ApplicationJob
  queue_as :background
  retry_on StandardError, wait: :polynomially_longer, attempts: 5

  # DHLT-05: Named constant for health check cadence (referenced by recurring.yml schedule).
  HEALTH_CHECK_INTERVAL = 30.minutes
  # DHLT-02: Number of consecutive failures before auto-phasing-out.
  MAX_CONSECUTIVE_FAILURES = 3

  def perform
    # DHLT-01: Check active CUSTOM domains only (subdomains are always valid, never checked).
    BrokerageDomain.without_deleted
                   .domain_type_custom
                   .status_active
                   .in_batches do |batch|
      batch.each { |domain| process_health_check(domain) }
    end
  end

  private

  def process_health_check(domain)
    if cname_resolves_to_subdomain?(domain)
      # Success: reset failure counter, update last checked
      domain.update!(consecutive_failures: 0, last_checked_at: Time.current)
    else
      # Failure: increment counter
      new_count = domain.consecutive_failures + 1
      domain.update!(consecutive_failures: new_count, last_checked_at: Time.current)

      # DHLT-02: After MAX_CONSECUTIVE_FAILURES, phase out and notify owners
      if new_count >= MAX_CONSECUTIVE_FAILURES
        domain.phase_out!
        # DHLT-03, DHLT-04: health_failure email — template handles primary fallback language
        DomainMailer.health_failure(domain).deliver_now!
      end
    end
  end

  # D-01: Use getresources (plural) — returns [] on miss, does not raise.
  # D-02: Target is the brokerage's active subdomain, looked up live.
  # D-03: Any failure (wrong target, error, timeout) returns false.
  def cname_resolves_to_subdomain?(domain)
    target = domain.brokerage.brokerage_domains
                             .domain_type_subdomain
                             .status_active
                             .first
                             &.domain
    return false if target.nil?

    Resolv::DNS.open do |dns|
      records = dns.getresources(domain.domain, Resolv::DNS::Resource::IN::CNAME)
      records.any? { |r| r.name.to_s.chomp(".") == target.chomp(".") }
    end
  rescue Resolv::ResolvError, Resolv::ResolvTimeout
    false
  end
end
