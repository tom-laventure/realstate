Create `backend/app/services/calendar/microsoft_calendar_adapter.rb`:

```ruby
module Calendar
  class MicrosoftCalendarAdapter
    GRAPH_BASE = "https://graph.microsoft.com/v1.0"

    def initialize(connection)
      @connection = connection
    end

    def sync(estate_date:, action:, external_event_id:, timezone:, credentials:)
      token = credentials["access_token"]

      case action
      when "create"
        HTTParty.post(
          "#{GRAPH_BASE}/me/events",
          headers: {
            "Authorization" => "Bearer #{token}",
            "Content-Type" => "application/json"
          },
          body: build_event_payload(estate_date, timezone, external_event_id).to_json
        ).tap do |response|
          raise "Microsoft Graph error #{response.code}: #{response.body}" unless response.success?
        end
      when "destroy"
        record = EstateDateCalendarEvent.find_by!(
          estate_date_id: estate_date.id,
          user_id: @connection.user_id,
          provider: "microsoft"
        )
        HTTParty.delete(
          "#{GRAPH_BASE}/me/events/#{record.external_event_id}",
          headers: {"Authorization" => "Bearer #{token}"}
        ).tap do |response|
          # 404 means already deleted — treat as success
          raise "Microsoft Graph error #{response.code}" unless response.success? || response.code == 404
        end
      else
        raise ArgumentError, "Unknown action: #{action}"
      end
    end

    private

    def build_event_payload(estate_date, timezone, external_event_id)
      tz = ActiveSupport::TimeZone[timezone] || ActiveSupport::TimeZone["UTC"]
      start_time = estate_date.date.in_time_zone(tz)
      end_time = start_time + 1.hour

      # Microsoft Graph requires ISO8601 without timezone offset — the timeZone field carries that
      {
        subject: estate_date.title.presence || estate_date.event_type,
        body: {
          contentType: "text",
          content: estate_date.notes.to_s
        },
        start: {
          dateTime: start_time.strftime("%Y-%m-%dT%H:%M:%S"),
          timeZone: timezone
        },
        end: {
          dateTime: end_time.strftime("%Y-%m-%dT%H:%M:%S"),
          timeZone: timezone
        },
        location: {
          displayName: estate_date.location.to_s
        },
        transactionId: external_event_id
      }
    end
  end
end
```

Key notes:
- `microsoft_graph` SDK is ARCHIVED — use HTTParty directly per RESEARCH.md "Alternatives Considered"
- `transactionId` field: Microsoft Graph idempotency key — set to `external_event_id` ("rs-{estate_date_id}") for safe retries
- For destroy: uses `record.external_event_id` (the Microsoft event ID stored during create) — NOT the `external_event_id` argument (which is just the deterministic "rs-N" string, not the actual Microsoft ID)
- 404 on delete is treated as success (already deleted)

```ruby
module V1
  module Account
    module Calendar
      class MicrosoftOauthController < ApplicationController
        include CognitoAuth
        include BrokerageAuth

        before_action :authenticate_cognito_user!

        # GET /v1/account/calendar/microsoft/authorize
        def authorize
          state = SecureRandom.hex(32)
          Rails.cache.write(
            "cal_oauth_state:#{state}",
            {user_id: Current.user.id},
            expires_in: 10.minutes
          )

          params_hash = {
            client_id: ENV.fetch("MICROSOFT_CALENDAR_CLIENT_ID"),
            redirect_uri: ENV.fetch("MICROSOFT_CALENDAR_REDIRECT_URI"),
            response_type: "code",
            scope: "https://graph.microsoft.com/Calendars.ReadWrite offline_access",
            state: state
          }

          redirect_url = "https://login.microsoftonline.com/common/oauth2/v2.0/authorize?" + params_hash.to_query

          render json: {redirect_url: redirect_url}, status: :ok
        end

        # GET /v1/account/calendar/microsoft/callback?code=...&state=...
        def callback
          state_data = Rails.cache.read("cal_oauth_state:#{params[:state]}")

          if state_data.nil?
            return render json: {error: "Invalid or expired state parameter"}, status: :unprocessable_entity
          end

          Rails.cache.delete("cal_oauth_state:#{params[:state]}")

          user_id = state_data[:user_id]

          token_response = exchange_code_for_tokens(params[:code])

          unless token_response.success?
            return render json: {error: "Token exchange failed"}, status: :unprocessable_entity
          end

          token_data = token_response.parsed_response

          conn = UserCalendarConnection.without_deleted.find_or_initialize_by(
            user_id: user_id,
            provider: "microsoft"
          )

          conn.credentials_hash = {
            "access_token" => token_data["access_token"],
            "refresh_token" => token_data["refresh_token"] || conn.credentials_hash["refresh_token"],
            "expires_at" => (Time.current + token_data["expires_in"].to_i.seconds).iso8601
          }
          conn.status = "connected"
          conn.last_error = nil
          conn.save!

          frontend_origin = request.base_url.sub("api.", "app.").gsub(%r{/v1.*}, "")
          redirect_to "#{frontend_origin}/account", allow_other_host: true
        end

        private

        def exchange_code_for_tokens(code)
          HTTParty.post(
            "https://login.microsoftonline.com/common/oauth2/v2.0/token",
            body: {
              client_id: ENV.fetch("MICROSOFT_CALENDAR_CLIENT_ID"),
              client_secret: ENV.fetch("MICROSOFT_CALENDAR_CLIENT_SECRET"),
              redirect_uri: ENV.fetch("MICROSOFT_CALENDAR_REDIRECT_URI"),
              code: code,
              grant_type: "authorization_code",
              scope: "https://graph.microsoft.com/Calendars.ReadWrite offline_access"
            }
          )
        end
      end
    end
  end
end
```

Security: same pattern as GoogleOauthController — user_id from Rails.cache only, state is single-use and 10-min TTL, no user-controlled input touches user_id (T-05-03-01).

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Authorize endpoint | Authenticated user; state nonce generated server-side |
| Microsoft → Callback endpoint | Untrusted `code` and `state` from Microsoft redirect |
| Callback → UserCalendarConnection | Token exchange result written to encrypted credentials |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-03-01 | Spoofing | `callback` state param | mitigate | Identical to T-05-02-01: state from `Rails.cache` only; single-use; 10-minute TTL |
| T-05-03-02 | Information Disclosure | Microsoft token values in logs | mitigate | Token exchange errors log response code only (`response.code`), not the body or token values |
| T-05-03-03 | Tampering | Microsoft token expiry (Pitfall 6) | accept | Microsoft refresh tokens expire after 90 days of inactivity; when `invalid_grant` occurs during refresh (in CalendarSyncJob Plan 05), `status = 'error'` is set and surfaces in UI per CAL-07; user must reconnect — no automatic fix possible |
| T-05-03-04 | Tampering | `transactionId` in Graph event | mitigate | `transactionId: external_event_id` provides idempotency key for retries; Microsoft deduplicates requests with same transactionId within 1 hour |
</threat_model>
