Create `backend/app/jobs/calendar_sync_job.rb`:

**CRITICAL — do NOT add `status: "connected"` to the `find_by!` call (BLOCKER 4 fix):**
If the job fails and the rescue block sets `conn.status = "error"`, a subsequent retry using `find_by!(status: "connected")` raises `RecordNotFound`, which triggers `discard_on RecordNotFound` and permanently drops the job. No retries execute. Instead, find by `user_id` + `provider` only and raise a custom `ConnectionNotActiveError` for the disconnected case — this fires `retry_on StandardError` (or a specific `discard_on`) rather than the permanent `discard_on RecordNotFound`.

```ruby
class CalendarSyncJob < ApplicationJob
  queue_as :default
  retry_on StandardError, wait: :polynomially_longer, attempts: 3
  discard_on ActiveRecord::RecordNotFound

  # Custom error class for intentionally-disconnected connections.
  # Raised when conn.status == "disconnected" — discard immediately, do not retry,
  # because the user deliberately disconnected.
  class ConnectionNotActiveError < StandardError; end
  discard_on ConnectionNotActiveError

  def perform(user_id:, provider:, estate_date_id:, action:)
    # Find WITHOUT status filter — RecordNotFound means the record is truly gone (correct to discard)
    # An error-state connection IS found and the job proceeds to retry the sync (CAL-07 requirement)
    conn = UserCalendarConnection.without_deleted
      .find_by!(user_id: user_id, provider: provider)

    # Guard disconnected separately — user disconnected intentionally, discard the job
    if conn.status == "disconnected"
      raise CalendarSyncJob::ConnectionNotActiveError,
        "Calendar connection #{provider} for user #{user_id} is disconnected"
    end

    estate_date = EstateDate.without_deleted.find(estate_date_id)
    timezone = estate_date.group_estate.group.brokerage.timezone.presence || "UTC"

    # Advisory lock prevents concurrent token refresh storms (CAL-06, Pitfall 3)
    UserCalendarConnection.with_advisory_lock("ucc-token-#{conn.id}") do
      refresh_token_if_expiring!(conn)
    end
    conn.reload

    # Deterministic provider event ID — "rs-{estate_date_id}" (Claude's Discretion, RESEARCH.md)
    external_event_id = "rs-#{estate_date_id}"

    adapter = adapter_for(conn)
    result = adapter.sync(
      estate_date: estate_date,
      action: action,
      external_event_id: external_event_id,
      timezone: timezone,
      credentials: conn.credentials_hash
    )

    # For Apple create, result is the CalDAV URL — store that as external_event_id (Pitfall 2)
    stored_event_id = (conn.provider == "apple" && action == "create" && result.is_a?(String)) ? result : external_event_id

    upsert_sync_record(
      estate_date_id: estate_date_id,
      user_id: user_id,
      provider: provider,
      external_event_id: stored_event_id,
      action: action
    )

    # Clear any prior error state on success
    conn.update_columns(last_error: nil)

  rescue => e
    # On any error: mark connection as error, store last_error, then re-raise for retry_on
    error_message = "#{e.class}: #{e.message.to_s.truncate(500)}"
    conn&.update_columns(status: "error", last_error: error_message)

    EstateDateCalendarEvent.where(
      estate_date_id: estate_date_id,
      user_id: user_id,
      provider: provider
    ).update_all(status: "error", last_error: error_message)

    raise
  end

  private

  def adapter_for(conn)
    case conn.provider
    when "google"    then Calendar::GoogleCalendarAdapter.new(conn)
    when "microsoft" then Calendar::MicrosoftCalendarAdapter.new(conn)
    when "apple"     then Calendar::AppleCalendarAdapter.new(conn)
    else raise ArgumentError, "Unknown provider: #{conn.provider}"
    end
  end

  def refresh_token_if_expiring!(conn)
    creds = conn.credentials_hash
    expires_at = creds["expires_at"]
    return unless expires_at

    expiry = Time.parse(expires_at) rescue nil
    return unless expiry
    return unless expiry < 5.minutes.from_now

    case conn.provider
    when "google"    then refresh_google_token!(conn)
    when "microsoft" then refresh_microsoft_token!(conn)
    # Apple: no token to refresh — app-specific password doesn't expire
    end
  end

  def refresh_google_token!(conn)
    creds = conn.credentials_hash
    response = HTTParty.post(
      "https://oauth2.googleapis.com/token",
      body: {
        client_id: ENV.fetch("GOOGLE_CALENDAR_CLIENT_ID"),
        client_secret: ENV.fetch("GOOGLE_CALENDAR_CLIENT_SECRET"),
        refresh_token: creds["refresh_token"],
        grant_type: "refresh_token"
      }
    )
    raise "Google token refresh failed: #{response.code}" unless response.success?

    data = response.parsed_response
    conn.credentials_hash = creds.merge(
      "access_token" => data["access_token"],
      "expires_at" => (Time.current + data["expires_in"].to_i.seconds).iso8601
    )
    conn.save!
  end

  def refresh_microsoft_token!(conn)
    creds = conn.credentials_hash
    response = HTTParty.post(
      "https://login.microsoftonline.com/common/oauth2/v2.0/token",
      body: {
        client_id: ENV.fetch("MICROSOFT_CALENDAR_CLIENT_ID"),
        client_secret: ENV.fetch("MICROSOFT_CALENDAR_CLIENT_SECRET"),
        refresh_token: creds["refresh_token"],
        grant_type: "refresh_token",
        scope: "https://graph.microsoft.com/Calendars.ReadWrite offline_access"
      }
    )
    raise "Microsoft token refresh failed: #{response.code}" unless response.success?

    data = response.parsed_response
    conn.credentials_hash = creds.merge(
      "access_token" => data["access_token"],
      "refresh_token" => data["refresh_token"] || creds["refresh_token"],
      "expires_at" => (Time.current + data["expires_in"].to_i.seconds).iso8601
    )
    conn.save!
  end

  def upsert_sync_record(estate_date_id:, user_id:, provider:, external_event_id:, action:)
    if action == "destroy"
      EstateDateCalendarEvent.where(
        estate_date_id: estate_date_id,
        user_id: user_id,
        provider: provider
      ).delete_all
    else
      EstateDateCalendarEvent.upsert(
        {
          estate_date_id: estate_date_id,
          user_id: user_id,
          provider: provider,
          external_event_id: external_event_id,
          status: "synced",
          last_error: nil,
          last_synced_at: Time.current,
          created_at: Time.current,
          updated_at: Time.current
        },
        unique_by: :idx_edce_unique,
        update_only: [:external_event_id, :status, :last_error, :last_synced_at, :updated_at]
      )
    end
  end
end
```

**Key design decisions:**
- `find_by!(user_id:, provider:)` WITHOUT `status:` filter — ensures retries work after error-state transitions (BLOCKER 4)
- `ConnectionNotActiveError` is a custom error class that triggers `discard_on ConnectionNotActiveError` — not `discard_on RecordNotFound` — for intentionally-disconnected connections
- `rescue => e` at the top level catches any provider error, updates `conn.status = 'error'`, then re-raises so `retry_on StandardError` can retry (CAL-07 requirement: sync failure visible in UI)
- `update_columns` used for error state updates (skips AR callbacks and validations — safe for status/error fields)
- `upsert` with `unique_by: :idx_edce_unique` for atomic create-or-update on estate_date_calendar_events
- Token refresh error logs `"#{e.class}: #{e.message.truncate(500)}"` — class + message but message is truncated and does NOT log raw token values

**Note on D-06 action values:** CONTEXT.md D-06 listed `'update'` as a potential action but `EstateEventsController` only has routes `[:index, :create, :destroy]` — no update action exists. `CalendarSyncJob` action values are `'create'` and `'destroy'` only. The `'update'` mention in D-06 was a listing artifact; the routes confirm the correct set.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| SolidQueue worker → CalendarSyncJob | Job arguments are integers and strings from Rails queue — not user-controlled at execution time |
| CalendarSyncJob → External APIs | Google/Microsoft Graph/Apple CalDAV — token sent in Authorization header |
| CalendarSyncJob → DB | Writes to estate_date_calendar_events and user_calendar_connections |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-05-01 | Tampering | Token refresh TOCTOU race | mitigate | `UserCalendarConnection.with_advisory_lock("ucc-token-#{conn.id}")` serializes all refresh operations for the same connection across all SolidQueue workers (CAL-06 requirement) |
| T-05-05-02 | Information Disclosure | Token values in job error logs | mitigate | `error_message = "#{e.class}: #{e.message.truncate(500)}"` — message truncated and never includes raw token strings; `refresh_google_token!` / `refresh_microsoft_token!` log only response.code on failure |
| T-05-05-03 | Denial of Service | Fan-out generates excessive jobs | accept | Fan-out is bounded by brokerage user count × 3 providers; typical brokerage < 100 users = < 300 jobs per estate event create — acceptable for SolidQueue :default queue |
| T-05-05-04 | Tampering | Soft-delete bypass in fan-out | mitigate | `Current.brokerage.users.without_deleted.pluck(:id)` and `UserCalendarConnection.without_deleted` per CLAUDE.md; deleted users' calendars are never updated |
| T-05-05-05 | Tampering | Apple CalDAV event URL stored incorrectly | mitigate | Apple `create` return value checked with `result.is_a?(String)` before using as stored_event_id; if non-String returned, falls back to `external_event_id` ("rs-N") string |
</threat_model>
