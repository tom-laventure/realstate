const ProviderCard: React.FC<ProviderCardProps> = ({
    connection,
    providerName,
    providerLogo,
    onConnect,
    onConnectApple,
    onDisconnect,
    onResync,
    isConnecting,
    isDisconnecting,
    isResyncing
}) => {
    const [disconnectDialogOpen, setDisconnectDialogOpen] = useState(false)
    const { status, lastSyncedAt, lastError, provider } = connection

    const formatLastSync = (ts: string | null) => {
        if (!ts) return 'Never synced'
        const diff = Date.now() - new Date(ts).getTime()
        const minutes = Math.floor(diff / 60000)
        const hours = Math.floor(minutes / 60)
        const days = Math.floor(hours / 24)
        if (days > 0) return `Last synced ${days} day${days > 1 ? 's' : ''} ago`
        if (hours > 0) return `Last synced ${hours} hour${hours > 1 ? 's' : ''} ago`
        if (minutes > 0) return `Last synced ${minutes} minute${minutes > 1 ? 's' : ''} ago`
        return 'Last synced just now'
    }

    const handleConfirmDisconnect = () => {
        onDisconnect(provider)
        setDisconnectDialogOpen(false)
    }

    return (
        <>
            <Card className="flex flex-col">
                <CardContent className="flex h-full flex-col gap-4 p-4 sm:p-6">
                    {/* Header row */}
                    <div className="flex items-center gap-2">
                        {providerLogo}
                        <span className="text-base font-semibold">{providerName}</span>
                    </div>

                    {/* Status row */}
                    <div className="flex items-center gap-2">
                        <StatusBadge status={status} />
                    </div>

                    {/* Last sync (not shown when disconnected) */}
                    {status !== 'disconnected' && (
                        <p className="text-xs text-muted-foreground">
                            {formatLastSync(lastSyncedAt)}
                        </p>
                    )}

                    {/* Error message (only when error) */}
                    {status === 'error' && lastError && (
                        <p className="text-xs text-muted-foreground">
                            Last sync failed. Click Re-sync to retry, or disconnect and reconnect if the problem persists.
                        </p>
                    )}

                    {/* Spacer pushes buttons to bottom */}
                    <div className="flex-1" />

                    {/* Action buttons */}
                    <div className="flex flex-col gap-2">
                        {status === 'connected' && (
                            <Button
                                variant="outline"
                                size="sm"
                                disabled={isDisconnecting}
                                onClick={() => setDisconnectDialogOpen(true)}
                                aria-label={`Disconnect ${providerName}`}
                                aria-busy={isDisconnecting}
                            >
                                {isDisconnecting
                                    ? <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden="true" />
                                    : <Unlink2 className="mr-2 h-4 w-4" aria-hidden="true" />
                                }
                                Disconnect
                            </Button>
                        )}

                        {status === 'error' && (
                            <>
                                <Button
                                    variant="default"
                                    size="sm"
                                    disabled={isResyncing}
                                    onClick={() => onResync(provider)}
                                    aria-label={`Re-sync ${providerName}`}
                                    aria-busy={isResyncing}
                                >
                                    <RefreshCw
                                        className={`mr-2 h-4 w-4 ${isResyncing ? 'animate-spin' : ''}`}
                                        aria-hidden="true"
                                    />
                                    Re-sync
                                </Button>
                                <Button
                                    variant="outline"
                                    size="sm"
                                    disabled={isDisconnecting}
                                    onClick={() => setDisconnectDialogOpen(true)}
                                    aria-label={`Disconnect ${providerName}`}
                                    aria-busy={isDisconnecting}
                                >
                                    {isDisconnecting
                                        ? <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden="true" />
                                        : <Unlink2 className="mr-2 h-4 w-4" aria-hidden="true" />
                                    }
                                    Disconnect
                                </Button>
                            </>
                        )}

                        {status === 'disconnected' && provider !== 'apple' && (
                            <Button
                                variant="default"
                                size="sm"
                                disabled={isConnecting}
                                onClick={onConnect}
                                aria-label={`Connect ${providerName}`}
                                aria-busy={isConnecting}
                            >
                                {isConnecting
                                    ? <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden="true" />
                                    : <Link2 className="mr-2 h-4 w-4" aria-hidden="true" />
                                }
                                Connect
                            </Button>
                        )}

                        {status === 'disconnected' && provider === 'apple' && (
                            <AppleCredentialForm
                                onSubmit={onConnectApple!}
                                isLoading={isConnecting}
                            />
                        )}
                    </div>
                </CardContent>
            </Card>

            {/* Disconnect confirmation dialog */}
            <Dialog open={disconnectDialogOpen} onOpenChange={setDisconnectDialogOpen}>
                <DialogContent>
                    <DialogHeader>
                        <DialogTitle>Disconnect {providerName}</DialogTitle>
                        <DialogDescription>
                            Your calendar will stop receiving new property showing events. Past events already synced will remain in your calendar.
                        </DialogDescription>
                    </DialogHeader>
                    <DialogFooter>
                        <DialogClose asChild>
                            <Button variant="outline">Keep Connected</Button>
                        </DialogClose>
                        <Button
                            variant="destructive"
                            disabled={isDisconnecting}
                            onClick={handleConfirmDisconnect}
                            aria-busy={isDisconnecting}
                        >
                            {isDisconnecting && <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden="true" />}
                            Disconnect
                        </Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>
        </>
    )
}
