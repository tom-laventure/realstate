Create `backend/app/services/calendar/apple_calendar_adapter.rb` (per CAL-05, RESEARCH.md Pattern 5):

```ruby
require "calendav"

module Calendar
  class AppleCalendarAdapter
    def initialize(connection)
      @connection = connection
    end

    def sync(estate_date:, action:, external_event_id:, timezone:, credentials:)
      creds = Calendav.credentials(
        :apple,
        credentials["email"],
        credentials["app_specific_password"]
      )
      client = Calendav.client(creds)

      case action
      when "create"
        calendar = client.calendars.list.first
        raise "No Apple calendar found for user" if calendar.nil?

        ics = build_ics(estate_date, external_event_id, timezone)
        # calendav returns the event URL from create — store it as external_event_id
        event_url = client.events.create(calendar.url, "#{external_event_id}.ics", ics.to_ical)
        # Caller (CalendarSyncJob) must use returned URL as external_event_id in upsert_sync_record
        # Return the URL so CalendarSyncJob can persist it
        event_url
      when "destroy"
        # For Apple, external_event_id stored in estate_date_calendar_events IS the URL (not "rs-N")
        record = EstateDateCalendarEvent.find_by!(
          estate_date_id: estate_date.id,
          user_id: @connection.user_id,
          provider: "apple"
        )
        client.events.delete(record.external_event_id)
        nil
      else
        raise ArgumentError, "Unknown action: #{action}"
      end
    end

    private

    def build_ics(estate_date, uid, timezone)
      cal = Icalendar::Calendar.new
      tz_obj = ActiveSupport::TimeZone[timezone] || ActiveSupport::TimeZone["UTC"]

      begin
        cal.add_timezone(tz_obj.tzinfo.ical_timezone)
      rescue
        # tzinfo timezone unavailable — proceed without VTIMEZONE component
      end

      event = Icalendar::Event.new
      event.uid = uid
      event.summary = estate_date.title.presence || estate_date.event_type
      event.location = estate_date.location.to_s
      event.description = estate_date.notes.to_s

      start_time = estate_date.date.in_time_zone(tz_obj)
      end_time = start_time + 1.hour

      event.dtstart = Icalendar::Values::DateTime.new(start_time, "TZID" => timezone)
      event.dtend   = Icalendar::Values::DateTime.new(end_time,   "TZID" => timezone)

      cal.add_event(event)
      cal
    end
  end
end
```

**Critical Apple pitfall (RESEARCH.md Pitfall 2):** For `create`, the return value of `client.events.create` is the CalDAV resource URL. CalendarSyncJob MUST store this URL (not the `"rs-N"` string) as `external_event_id` in `estate_date_calendar_events` for Apple. The `sync` method returns the URL for `create` action; `nil` for `destroy`. CalendarSyncJob will use the return value to store the correct external_event_id.

Create `backend/app/controllers/v1/account/calendar/connections_controller.rb`:

```ruby
module V1
  module Account
    module Calendar
      class ConnectionsController < ApplicationController
        include CognitoAuth
        include BrokerageAuth

        before_action :authenticate_cognito_user!

        PROVIDERS = %w[google microsoft apple].freeze

        # GET /v1/account/calendar/connections
        # Returns all 3 provider connections (creates missing ones as 'disconnected')
        def index
          # Ensure all 3 provider rows exist for the user (upsert-style read)
          connections = PROVIDERS.map do |provider|
            Current.user.user_calendar_connections.without_deleted.find_or_create_by!(
              provider: provider
            ) do |conn|
              conn.status = "disconnected"
            end
          end

          render json: {
            connections: ActiveModelSerializers::SerializableResource.new(
              connections,
              each_serializer: UserCalendarConnectionSerializer,
              adapter: :attributes
            ).as_json
          }, status: :ok
        end

        # POST /v1/account/calendar/apple/connect
        # Validates Apple credentials (CalDAV HEAD check) and stores them
        def create_apple
          email = params[:email].to_s.strip
          app_specific_password = params[:app_specific_password].to_s.strip

          if email.blank? || app_specific_password.blank?
            return render json: {error: "Email and app_specific_password are required"}, status: :unprocessable_entity
          end

          # Validate credentials before storing by attempting to list calendars
          begin
            creds = Calendav.credentials(:apple, email, app_specific_password)
            client = Calendav.client(creds)
            client.calendars.list  # raises on invalid credentials
          rescue => e
            return render json: {
              error: "Connection failed. Check your iCloud email and app-specific password, then try again."
            }, status: :unprocessable_entity
          end

          conn = Current.user.user_calendar_connections.without_deleted.find_or_initialize_by(provider: "apple")
          conn.credentials_hash = {
            "email" => email,
            "app_specific_password" => app_specific_password
          }
          conn.status = "connected"
          conn.last_error = nil
          conn.save!

          render json: {
            connection: UserCalendarConnectionSerializer.new(conn).as_json
          }, status: :ok
        end

        # DELETE /v1/account/calendar/:provider
        # Disconnects a provider (status=disconnected, credentials cleared)
        def destroy
          unless PROVIDERS.include?(params[:provider])
            return render json: {error: "Invalid provider"}, status: :unprocessable_entity
          end

          conn = Current.user.user_calendar_connections.without_deleted.find_by!(provider: params[:provider])
          conn.update!(
            status: "disconnected",
            credentials: nil,
            last_error: nil
          )

          render json: {}, status: :ok
        end

        # POST /v1/account/calendar/:provider/resync
        # Enqueues CalendarSyncJob for all future EstateDate records in user's brokerage (D-11)
        def resync
          unless PROVIDERS.include?(params[:provider])
            return render json: {error: "Invalid provider"}, status: :unprocessable_entity
          end

          conn = Current.user.user_calendar_connections.without_deleted.find_by!(
            provider: params[:provider],
            status: "connected"
          )

          if conn.nil?
            return render json: {error: "No connected #{params[:provider]} calendar"}, status: :unprocessable_entity
          end

          # Full sweep of future EstateDate records in user's brokerage — Pitfall 8
          future_estate_dates = EstateDate.without_deleted
            .joins(group_estate: {group: :brokerage})
            .where(groups: {brokerage_id: Current.brokerage.id})
            .where("estate_dates.date >= ?", Time.current)
            .where(estates: {deleted_at: nil})

          future_estate_dates.find_each do |estate_date|
            CalendarSyncJob.perform_later(
              user_id: Current.user.id,
              provider: params[:provider],
              estate_date_id: estate_date.id,
              action: "create"
            )
          end

          render json: {enqueued: future_estate_dates.count}, status: :ok
        end
      end
    end
  end
end
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-04-01 | Tampering | `params[:provider]` in destroy/resync | mitigate | `PROVIDERS.include?(params[:provider])` guard returns 422 on invalid values; never use provider string in SQL without validation |
| T-05-04-02 | Information Disclosure | `UserCalendarConnectionSerializer` | mitigate | Attributes whitelist: `id, provider, status, last_synced_at, last_error, created_at` — credentials never serialized; enforced by omitting :credentials from attributes list |
| T-05-04-03 | Tampering | Apple credentials brute-force on connect | mitigate | Live CalDAV validation on `create_apple`; failed credential check returns 422 immediately; no retry loop in controller |
| T-05-04-04 | Elevation of Privilege | Cross-user connection access | mitigate | All queries scoped to `Current.user.user_calendar_connections` — never `UserCalendarConnection.find(params[:id])` |
| T-05-04-05 | Information Disclosure | Apple app-specific password in error logs | mitigate | `create_apple` rescue block logs exception class only (implicit — `e` message may not contain password, but do not log `e.message` for Apple auth errors); return generic error message to client |
| T-05-04-06 | Tampering | Resync scope bypass (Pitfall 8) | mitigate | Resync scoped to `estate_dates.date >= Time.current` — past events never re-synced; also guards `estates: {deleted_at: nil}` per CLAUDE.md soft-delete requirement |
