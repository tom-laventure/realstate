Create the `backend/app/services/calendar/` directory if it does not exist, then create `google_calendar_adapter.rb`:

```ruby
require "google/apis/calendar_v3"
require "googleauth"

module Calendar
  class GoogleCalendarAdapter
    CALENDAR_ID = "primary"

    def initialize(connection)
      @connection = connection
    end

    def sync(estate_date:, action:, external_event_id:, timezone:, credentials:)
      service = Google::Apis::CalendarV3::CalendarService.new
      service.authorization = build_authorizer(credentials)

      case action
      when "create"
        event = build_event(estate_date, timezone)
        event.id = external_event_id
        service.insert_event(CALENDAR_ID, event)
      when "destroy"
        service.delete_event(CALENDAR_ID, external_event_id)
      else
        raise ArgumentError, "Unknown action: #{action}"
      end
    end

    private

    def build_authorizer(creds)
      Signet::OAuth2::Client.new(
        client_id: ENV.fetch("GOOGLE_CALENDAR_CLIENT_ID"),
        client_secret: ENV.fetch("GOOGLE_CALENDAR_CLIENT_SECRET"),
        access_token: creds["access_token"],
        refresh_token: creds["refresh_token"],
        expires_at: creds["expires_at"]
      )
    end

    def build_event(estate_date, timezone)
      tz = ActiveSupport::TimeZone[timezone] || ActiveSupport::TimeZone["UTC"]
      start_time = estate_date.date.in_time_zone(tz)
      end_time = start_time + 1.hour

      Google::Apis::CalendarV3::Event.new(
        summary: estate_date.title.presence || estate_date.event_type,
        location: estate_date.location,
        description: estate_date.notes,
        start: Google::Apis::CalendarV3::EventDateTime.new(
          date_time: start_time.rfc3339,
          time_zone: timezone
        ),
        end: Google::Apis::CalendarV3::EventDateTime.new(
          date_time: end_time.rfc3339,
          time_zone: timezone
        )
      )
    end
  end
end
```

Key points:
- `estate_date.date.in_time_zone(tz)` converts UTC datetime to brokerage timezone (per D-04, Pitfall 7)
- Adapter does NOT do token refresh — that is CalendarSyncJob's responsibility with advisory lock (per D-06, CAL-06)
- `external_event_id` is the deterministic `"rs-{estate_date_id}"` string passed in by the job
- `ENV.fetch("GOOGLE_CALENDAR_CLIENT_ID")` — will raise KeyError if missing, surfacing configuration problems clearly

Create `backend/app/controllers/v1/account/calendar/google_oauth_controller.rb`:

```ruby
module V1
  module Account
    module Calendar
      class GoogleOauthController < ApplicationController
        include CognitoAuth
        include BrokerageAuth

        before_action :authenticate_cognito_user!

        # GET /v1/account/calendar/google/authorize
        # Returns { redirect_url: "https://accounts.google.com/..." }
        # Frontend calls window.location.href = data.redirectUrl after receiving this
        def authorize
          state = SecureRandom.hex(32)
          Rails.cache.write(
            "cal_oauth_state:#{state}",
            {user_id: Current.user.id},
            expires_in: 10.minutes
          )

          params_hash = {
            client_id: ENV.fetch("GOOGLE_CALENDAR_CLIENT_ID"),
            redirect_uri: ENV.fetch("GOOGLE_CALENDAR_REDIRECT_URI"),
            response_type: "code",
            scope: "https://www.googleapis.com/auth/calendar.events",
            access_type: "offline",
            prompt: "consent",
            state: state
          }

          redirect_url = "https://accounts.google.com/o/oauth2/v2/auth?" + params_hash.to_query

          render json: {redirect_url: redirect_url}, status: :ok
        end

        # GET /v1/account/calendar/google/callback?code=...&state=...
        # Called by Google after user authorizes. Exchanges code for tokens,
        # upserts UserCalendarConnection, redirects browser back to frontend /account.
        def callback
          state_data = Rails.cache.read("cal_oauth_state:#{params[:state]}")

          if state_data.nil?
            return render json: {error: "Invalid or expired state parameter"}, status: :unprocessable_entity
          end

          Rails.cache.delete("cal_oauth_state:#{params[:state]}")

          user_id = state_data[:user_id]

          token_response = exchange_code_for_tokens(params[:code])

          unless token_response.success?
            return render json: {error: "Token exchange failed"}, status: :unprocessable_entity
          end

          token_data = token_response.parsed_response

          conn = UserCalendarConnection.without_deleted.find_or_initialize_by(
            user_id: user_id,
            provider: "google"
          )

          conn.credentials_hash = {
            "access_token" => token_data["access_token"],
            "refresh_token" => token_data["refresh_token"] || conn.credentials_hash["refresh_token"],
            "expires_at" => (Time.current + token_data["expires_in"].to_i.seconds).iso8601
          }
          conn.status = "connected"
          conn.last_error = nil
          conn.save!

          # Redirect browser back to frontend account page
          frontend_origin = request.base_url.sub("api.", "app.").gsub(%r{/v1.*}, "")
          redirect_to "#{frontend_origin}/account", allow_other_host: true
        end

        private

        def exchange_code_for_tokens(code)
          HTTParty.post(
            "https://oauth2.googleapis.com/token",
            body: {
              code: code,
              client_id: ENV.fetch("GOOGLE_CALENDAR_CLIENT_ID"),
              client_secret: ENV.fetch("GOOGLE_CALENDAR_CLIENT_SECRET"),
              redirect_uri: ENV.fetch("GOOGLE_CALENDAR_REDIRECT_URI"),
              grant_type: "authorization_code"
            }
          )
        end
      end
    end
  end
end
```

**Security notes implemented:**
- State parameter read from `Rails.cache` — never from `params` or user input for user_id (per T-05-02-01 / Pitfall 1)
- State is single-use: deleted from cache immediately after reading
- `refresh_token` preserved from existing connection if Google does not re-issue it (Google only re-issues on first `prompt: consent`)
- `allow_other_host: true` is required because the redirect crosses subdomains (brokerage subdomain backend → frontend)

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Authorize endpoint | Authenticated user initiates OAuth; state nonce generated server-side |
| Google → Callback endpoint | Untrusted `code` and `state` params arrive from Google redirect |
| Callback → UserCalendarConnection | Token exchange result written to encrypted credentials field |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-02-01 | Spoofing | `callback` state param | mitigate | `state` validated against `Rails.cache` entry; entry deleted on first read (single-use); expired (>10 min) state returns 422 |
| T-05-02-02 | Information Disclosure | Token exchange response | mitigate | `exchange_code_for_tokens` logs only exception class if it raises; token values never written to Rails.logger |
| T-05-02-03 | Tampering | `UserCalendarConnection` user_id | mitigate | `user_id` taken from `state_data[:user_id]` (Rails.cache, server-set at authorize time) — never from `params[:user_id]`; cannot be tampered by attacker controlling query string |
| T-05-02-04 | Information Disclosure | GoogleCalendarAdapter credentials | mitigate | Adapter receives `credentials` Hash (already decrypted by AR Encryption in CalendarSyncJob); does not write to logs; `ENV.fetch` raises loudly on misconfiguration rather than silently sending nil |
| T-05-02-05 | Elevation of Privilege | OAuth callback without Cognito auth | accept | `callback` action does NOT require `authenticate_cognito_user!` because it is the browser-redirect leg (no JWT available); user identity is established via `state` cache lookup — this is the standard OAuth2 pattern and is safe given single-use state with 10-min TTL |
</threat_model>
