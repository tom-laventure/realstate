    Step A — Create `frontend/src/components/view/admin/domainsAdminView/slugValidation.ts`:

    ```ts
    import { RESERVED_SLUGS } from '@/assets/types/domainType'

    export interface SlugValidationResult {
        isValid: boolean
        error: string | null
        isUnchanged: boolean
    }

    // Validation order matters (Pitfall 4 in 21-RESEARCH.md):
    // 1. Empty       → save disabled, no error shown
    // 2. Pattern     → leading/trailing hyphen and general-format messages distinguished
    // 3. Reserved    → reserved word message
    // 4. Unchanged   → save disabled, no error shown
    const SLUG_PATTERN = /^[a-z0-9]([a-z0-9-]*[a-z0-9])?$|^[a-z0-9]$/
    const HYPHEN_EDGE_PATTERN = /^-|-$/

    export const validateSlug = (
        slug: string,
        currentSlug: string,
        reservedSlugs: readonly string[] = RESERVED_SLUGS,
    ): SlugValidationResult => {
        // 1. Empty
        if (slug.length === 0) {
            return { isValid: false, error: null, isUnchanged: false }
        }

        // 2. Pattern: discriminate hyphen-edge from generic format
        if (HYPHEN_EDGE_PATTERN.test(slug)) {
            return { isValid: false, error: 'Slug cannot start or end with a hyphen.', isUnchanged: false }
        }
        if (!SLUG_PATTERN.test(slug)) {
            return { isValid: false, error: 'Slug must contain only letters, numbers, and hyphens.', isUnchanged: false }
        }

        // 3. Reserved
        if (reservedSlugs.includes(slug)) {
            return { isValid: false, error: `"${slug}" is a reserved name and cannot be used.`, isUnchanged: false }
        }

        // 4. Unchanged
        if (slug === currentSlug) {
            return { isValid: false, error: null, isUnchanged: true }
        }

        return { isValid: true, error: null, isUnchanged: false }
    }

    export default validateSlug
    ```

    import { describe, it, expect } from 'vitest'
    import { validateSlug } from './slugValidation'
    import { RESERVED_SLUGS } from '@/assets/types/domainType'

    describe('validateSlug', () => {
        const current = 'acme'

        it('returns not-valid with null error for empty input', () => {
            const result = validateSlug('', current)
            expect(result.isValid).toBe(false)
            expect(result.error).toBeNull()
            expect(result.isUnchanged).toBe(false)
        })

        it('returns format error for uppercase/space input', () => {
            const result = validateSlug('BAD CASE', current)
            expect(result.isValid).toBe(false)
            expect(result.error).toBe('Slug must contain only letters, numbers, and hyphens.')
        })

        it('returns format error for symbols', () => {
            expect(validateSlug('a_b', current).error).toBe('Slug must contain only letters, numbers, and hyphens.')
        })

        it('returns hyphen-edge error for leading hyphen', () => {
            expect(validateSlug('-acme', current).error).toBe('Slug cannot start or end with a hyphen.')
        })

        it('returns hyphen-edge error for trailing hyphen', () => {
            expect(validateSlug('acme-', current).error).toBe('Slug cannot start or end with a hyphen.')
        })

        it('returns reserved error with the slug quoted in the message', () => {
            expect(validateSlug('admin', current).error).toBe('"admin" is a reserved name and cannot be used.')
            expect(validateSlug('admin', current).isValid).toBe(false)
        })

        it('returns isUnchanged true (not valid, no error) when slug equals currentSlug', () => {
            const result = validateSlug('acme', 'acme')
            expect(result.isValid).toBe(false)
            expect(result.error).toBeNull()
            expect(result.isUnchanged).toBe(true)
        })

        it('accepts a valid single-character slug', () => {
            expect(validateSlug('a', current)).toEqual({ isValid: true, error: null, isUnchanged: false })
        })

        it('accepts a valid hyphenated slug', () => {
            expect(validateSlug('acme-new', current)).toEqual({ isValid: true, error: null, isUnchanged: false })
        })

        it('does not flag a non-reserved word that contains a reserved word as substring', () => {
            // 'apps' is NOT in RESERVED_SLUGS but 'app' is — must use exact match
            expect(RESERVED_SLUGS).toContain('app')
            expect(RESERVED_SLUGS).not.toContain('apps')
            expect(validateSlug('apps', current).isValid).toBe(true)
        })

        it('flags all 21 RESERVED_SLUGS entries individually', () => {
            for (const reserved of RESERVED_SLUGS) {
                const r = validateSlug(reserved, 'something-else')
                expect(r.isValid).toBe(false)
                expect(r.error).toBe(`"${reserved}" is a reserved name and cannot be used.`)

        })

    ```tsx
    import React from 'react'
    import { Input } from '@/components/ui/input'

    interface DomainSlugInputProps {
        value: string
        onChange: (value: string) => void
        baseDomain?: string
        error?: string | null
        disabled?: boolean
    }

    const normalize = (raw: string): string =>
        raw.toLowerCase().replace(/[^a-z0-9-]/g, '')

    const DomainSlugInput: React.FC<DomainSlugInputProps> = ({
        value,
        onChange,
        baseDomain = 'realstate.app',
        error,
        disabled,
    }) => {
        const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
            onChange(normalize(e.target.value))
        }

        return (
            <div className="space-y-1.5">
                <div className="flex items-center rounded-md border border-input bg-background focus-within:ring-1 focus-within:ring-ring">
                    <Input
                        value={value}
                        onChange={handleChange}
                        placeholder="your-brokerage"
                        disabled={disabled}
                        className="border-0 shadow-none focus-visible:ring-0 focus-visible:ring-offset-0"
                        maxLength={63}
                        aria-label="Subdomain slug"
                        aria-invalid={!!error}
                    />
                    <span className="shrink-0 select-none pr-3 text-sm text-muted-foreground">
                        .{baseDomain}
                    </span>
                </div>
                {error && <p className="text-xs text-destructive">{error}</p>}
            </div>
        )
    }

    export default DomainSlugInput
    ```

    NOTE: `normalize` here intentionally does NOT strip leading/trailing hyphens (unlike the original SubdomainInput) — the validation layer surfaces a hyphen-edge error rather than silently rewriting the user's input. This matches D-11's "validation on-change" + "Save button disabled while invalid" contract from 21-CONTEXT.md.

    import React, { FormEvent, useState } from 'react'
    import { Button } from '@/components/ui/button'
    import DomainSlugInput from './DomainSlugInput'
    import { validateSlug } from './slugValidation'

    interface SubdomainCardEditFormProps {
        // The current full domain (e.g. 'acme.realstate.app') — slug is the first label.
        currentDomain: string
        baseDomain?: string
        disabled?: boolean
        onSubmit: (slug: string) => void
        onCancel: () => void
    }

    const SubdomainCardEditForm: React.FC<SubdomainCardEditFormProps> = ({
        currentDomain,
        baseDomain = 'realstate.app',
        disabled,
        onSubmit,
        onCancel,

        const currentSlug = currentDomain.split('.')[0] ?? ''

        const { isValid, error } = validateSlug(value, currentSlug)

                    value={value}
                    onChange={setValue}
                    baseDomain={baseDomain}
                    error={error}
                    disabled={disabled}
                />
                <div className="flex items-center gap-2">

                        Save changes
                    </Button>
                    <Button type="button" variant="outline" onClick={onCancel} disabled={disabled}>
                        Discard changes
                    </Button>
                </div>

        const days = Math.max(0, dayjs(phaseOutAt).diff(dayjs(), 'day'))
        const date = dayjs(phaseOutAt).format('MMM D')
        return `${days} days remaining · decommissions ${date}`
    }

        onDecommission,

        const [removeDialogOpen, setRemoveDialogOpen] = useState(false)
        const isSubdomain = domain.domainType === 'subdomain'
        const isCustom = domain.domainType === 'custom'
        const isActive = domain.status === 'active'

                    )}
