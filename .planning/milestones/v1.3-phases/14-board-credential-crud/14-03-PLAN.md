    ```typescript
    import { useEffect, useState } from 'react'
    import { Info, Loader2 } from 'lucide-react'
    import {
        Dialog,
        DialogContent,
        DialogHeader,
        DialogTitle,
        DialogDescription
    } from '@/components/ui/dialog'
    import { Button } from '@/components/ui/button'
    import { Input } from '@/components/ui/input'
    import { Label } from '@/components/ui/label'
    import toast from 'react-hot-toast'
    import { BoardCredentialType, UpdateBoardCredentialPayload } from '@/assets/types/boardType'

    interface EditBoardCredentialDialogProps {
        open: boolean
        onOpenChange: (open: boolean) => void
        credential: BoardCredentialType | null
        onEdit: (id: string, payload: UpdateBoardCredentialPayload) => Promise<void>
    }

    const EditBoardCredentialDialog = ({
        open,
        onOpenChange,
        credential,
        onEdit
    }: EditBoardCredentialDialogProps) => {
        const [username, setUsername] = useState('')
        const [password, setPassword] = useState('')
        const [isSubmitting, setIsSubmitting] = useState(false)
        const [serverError, setServerError] = useState<string | null>(null)

        const isTestBoard = credential?.realEstateBoard.isTestBoard ?? false
        const boardName = credential?.realEstateBoard.name ?? ''

        // Reset transient state whenever the dialog closes or the target credential changes
        useEffect(() => {
            if (!open) {
                setUsername('')
                setPassword('')
                setServerError(null)
                setIsSubmitting(false)
            }
        }, [open, credential?.id])

        const handleOpenChange = (nextOpen: boolean) => {
            if (!nextOpen) {
                setUsername('')
                setPassword('')
                setServerError(null)
                setIsSubmitting(false)
            }
            onOpenChange(nextOpen)
        }

        const handleSubmit = async () => {
            if (!credential) return
            if (!isTestBoard && (!username || !password)) return
            setIsSubmitting(true)
            setServerError(null)
            try {
                const payload: UpdateBoardCredentialPayload = isTestBoard
                    ? {}
                    : { retsUsername: username, retsPassword: password }
                await onEdit(credential.id, payload)
                toast.success(isTestBoard ? 'Board connection verified.' : 'Board credentials updated.')
                handleOpenChange(false)
            } catch (err: unknown) {
                const axiosError = err as { response?: { data?: { error?: string } } }
                setServerError(axiosError?.response?.data?.error || 'Update failed. Please try again.')
            } finally {
                setIsSubmitting(false)
            }
        }

        const submitDisabled = isTestBoard
            ? isSubmitting
            : isSubmitting || !username || !password

        const submitLabel = isTestBoard ? 'Re-verify connection' : 'Save credentials'

        return (
            <Dialog open={open} onOpenChange={handleOpenChange}>
                <DialogContent className="sm:max-w-md">
                    <DialogHeader>
                        <DialogTitle>Edit Board Connection</DialogTitle>
                        <DialogDescription>
                            {isTestBoard
                                ? `Manage connection to ${boardName}.`
                                : `Re-enter RETS credentials for ${boardName}.`}
                        </DialogDescription>
                    </DialogHeader>

                    <div className="space-y-4 pt-2">
                        {isTestBoard ? (
                            <div className="rounded-md border border-blue-200 bg-blue-50 px-4 py-2 text-xs text-blue-800 flex items-start gap-2">
                                <Info className="h-4 w-4 mt-0.5 shrink-0" />
                                <span>Test board — no credentials required.</span>
                            </div>
                        ) : (
                            <>
                                <div className="space-y-2">
                                    <Label htmlFor="edit-rets-username">RETS Username</Label>
                                    <Input
                                        id="edit-rets-username"
                                        autoFocus
                                        value={username}
                                        onChange={e => setUsername(e.target.value)}
                                        placeholder="Enter your RETS username"
                                        autoComplete="off"
                                    />
                                </div>
                                <div className="space-y-2">
                                    <Label htmlFor="edit-rets-password">RETS Password</Label>
                                    <Input
                                        id="edit-rets-password"
                                        type="password"
                                        value={password}
                                        onChange={e => setPassword(e.target.value)}
                                        placeholder="Enter your RETS password"
                                        autoComplete="new-password"
                                    />
                                </div>
                            </>
                        )}

                        {serverError && (
                            <p className="text-sm text-destructive">{serverError}</p>
                        )}

                        <div className="flex items-center justify-end pt-1">
                            <Button
                                onClick={handleSubmit}
                                disabled={submitDisabled}
                                autoFocus={isTestBoard}
                            >
                                {isSubmitting && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
                                {submitLabel}
                            </Button>
                        </div>
                    </div>
                </DialogContent>
            </Dialog>
        )
    }

    export default EditBoardCredentialDialog
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser DOM → React state | User clicks Delete/Edit menu items; ⋯ button trigger crosses untrusted-input boundary |
| dialog → context.handleEditCredential | Form data crosses to mutation hook |
| context → axios → backend | Credentials in transit (HTTPS in prod; localhost in dev) |
| permission UI guard (`Can`) | NOT a security boundary — backend require_admin! is the real check (mitigated in Plan 01) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-03-01 | Information Disclosure | EditBoardCredentialDialog logging | mitigate | Component does NOT call `console.log` / `console.error` on credentials. Error path uses `setServerError(...)` with the backend's redacted error message; raw payload is never logged or interpolated into the DOM as a serializer output. Plain text password is held only in local `useState` and cleared on dialog close via `handleOpenChange`. |
| T-14-03-02 | Tampering | Frontend permission bypass | accept (transferred to Plan 01) | `Can permission="manage_brokerage_settings"` is UX-only — a malicious admin could remove the guard in devtools, but Plan 01's `require_admin!` + `Current.brokerage` scoping enforce server-side. Documented in CLAUDE.md ("Authorization is imperative — no declarative gem"). |
| T-14-03-03 | Tampering / IDOR | Delete confirmation passes wrong id | mitigate | Delete handler uses `deleteTarget.id` from the row's `credential` object (closure-scoped); not user-typed. AlertDialog state stores the full credential record, eliminating risk of confusing rows. Backend (Plan 01) re-validates `Current.brokerage` scope on every DELETE. |
| T-14-03-04 | Spoofing | Reading credential name from prop | accept | `credential.realEstateBoard.name` is rendered as plain text inside React JSX (auto-escaped). XSS via injected board name is not possible without React-level escape bypass; no `dangerouslySetInnerHTML` is used. |
| T-14-03-05 | Repudiation | No audit log for delete in UI | accept | UI does not need to display an audit trail. `acts_as_paranoid` (Plan 01) retains soft-deleted records server-side; a future admin audit view would query `.with_deleted`. Out of scope for this phase. |
| T-14-03-06 | Denial of Service | Spam-clicking Edit submit | mitigate | Submit button disabled while `isSubmitting === true`; double-click prevented. Backend has admin auth + brokerage scope, limiting blast radius to admin's own brokerage. |
| T-14-03-07 | Information Disclosure | Browser autofill leaking RETS credentials to password manager | accept | `autoComplete="new-password"` on the password input discourages autofill suggestion; `autoComplete="off"` on username. Password managers may still capture if user explicitly saves — acceptable risk; users with RETS credentials are aware they manage them. |
