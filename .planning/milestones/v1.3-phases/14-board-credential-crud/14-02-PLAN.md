    ```typescript
    import { useMutation, useQueryClient } from '@tanstack/react-query'
    import deleteBoardCredential from '@/assets/api/admin/boards/deleteBoardCredential'
    import { boardCredentialsKeys } from '@/store/hooks/queryKeys'
    import { AxiosErrorType } from '@/assets/types/axiosErrorType'
    import toast from 'react-hot-toast'

    const useDeleteBoardCredential = () => {
        const queryClient = useQueryClient()

        return useMutation({
            mutationFn: (id: string) => deleteBoardCredential(id),
            onSuccess: () => {
                queryClient.invalidateQueries({ queryKey: boardCredentialsKeys.all })
                toast.success('Board disconnected.')
            },
            onError: (err: AxiosErrorType) => {
                toast.error('Failed to disconnect board. Please try again.')
                console.error('Failed to delete board credential:', err)
            }
        })
    }

    export default useDeleteBoardCredential
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser hook → axiosInstance | Hook hands raw `payload` and `id` to API module, which forwards to backend |
| axiosInstance → backend | Backend re-validates everything (Plan 01 enforces all server-side checks) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-02-01 | Information Disclosure | Frontend logging of credentials | mitigate | `useDeleteBoardCredential` `console.error` logs only the AxiosError object (no payload — DELETE has no body). `useUpdateBoardCredential` has no onError handler at all, so credentials never reach `console.error` via the hook. Dialog (Plan 03) is responsible for not logging credentials. |
| T-14-02-02 | Tampering | Hook param shapes | mitigate | `useUpdateBoardCredential` mutationFn signature is `{ id: string; payload: UpdateBoardCredentialPayload }` — TypeScript prevents callers passing untyped objects. `UpdateBoardCredentialPayload` permits only optional `retsUsername` / `retsPassword` (no boardId), so the hook cannot be used to reassign a credential to another board even by mistake. |
| T-14-02-03 | Spoofing | shadcn install authenticity | accept | shadcn registry is trusted (used across the project). `npx shadcn@latest add alert-dialog` pulls from the official shadcn CDN; no third-party registry. Risk equivalent to all prior shadcn installs (button, dialog, etc.). |
| T-14-02-04 | Elevation of Privilege | Frontend cannot enforce authorization | mitigate | This plan does not attempt frontend authorization for these endpoints; backend (Plan 01) enforces `Current.brokerage` scoping + `require_admin!`. Frontend UI guard (`Can permission="manage_brokerage_settings"`) is a UX nicety in Plan 03; the security boundary is server-side. |
| T-14-02-05 | Information Disclosure | API module response shape | accept | `updateBoardCredential` returns `BoardCredentialType` which does NOT include `retsUsername` / `retsPassword` (verified from existing serializer + boardType.ts). Plaintext credentials never round-trip back to the client. |
</threat_model>
