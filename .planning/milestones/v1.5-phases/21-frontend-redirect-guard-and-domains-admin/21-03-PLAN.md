    Step A — Create `frontend/src/containers/adminBoard/domainsAdminBoard/useDomainsAdminContext.ts` with EXACTLY the content below. This is the brain. It computes `subdomainSlotsBlocked` and `customSlotsBlocked` per Pitfall 5 in 21-RESEARCH.md (the slot block must be a GLOBAL property, not per-card):

    ```ts
    import { useCallback, useMemo, useState } from 'react'
    import useFetchDomains from '@/store/hooks/Admin/Domains/useFetchDomains'
    import useFetchBrokerage from '@/store/hooks/Admin/Brokerage/useFetchBrokerage'
    import useUpdateSubdomain from '@/store/hooks/Admin/Domains/useUpdateSubdomain'
    import useSetPrimaryDomain from '@/store/hooks/Admin/Domains/useSetPrimaryDomain'
    import useDecommissionDomain from '@/store/hooks/Admin/Domains/useDecommissionDomain'
    import useAddCustomDomain from '@/store/hooks/Admin/Domains/useAddCustomDomain'
    import useRemoveCustomDomain from '@/store/hooks/Admin/Domains/useRemoveCustomDomain'
    import { PhaseOutWarning } from '@/assets/types/domainType'

    const useDomainsAdminContext = () => {
        // D-12 — ephemeral slot warning, reset on next page load.
        const [slotWarning, setSlotWarning] = useState<PhaseOutWarning | null>(null)

        const { data: domainsData, isLoading: isLoadingDomains, error: domainsError } = useFetchDomains()
        const { data: brokerage, isLoading: isLoadingBrokerage, error: brokerageError } = useFetchBrokerage()

        const { mutate: updateSubdomainMutation } = useUpdateSubdomain(setSlotWarning)
        const { mutate: setPrimaryMutation } = useSetPrimaryDomain()
        const { mutate: decommissionMutation } = useDecommissionDomain()
        const { mutate: addCustomMutation } = useAddCustomDomain(setSlotWarning)
        const { mutate: removeCustomMutation } = useRemoveCustomDomain(setSlotWarning)

        const updateSubdomain = useCallback((slug: string) => updateSubdomainMutation(slug), [updateSubdomainMutation])
        const setPrimary = useCallback((id: string | number) => setPrimaryMutation(id), [setPrimaryMutation])
        const decommission = useCallback((id: string | number) => decommissionMutation(id), [decommissionMutation])
        const addCustom = useCallback((domain: string) => addCustomMutation(domain), [addCustomMutation])
        const removeCustom = useCallback((id: string | number) => removeCustomMutation(id), [removeCustomMutation])

        const domains = domainsData?.domains ?? []

        // Pitfall 5 (21-RESEARCH.md): slot count is GLOBAL, not per-card.
        // 2 phasing-out of a type means 0 slots left for new actions of that type.
        const subdomainSlotsBlocked = domains.filter(
            d => d.domainType === 'subdomain' && d.status === 'phasing_out'
        ).length >= 2

        const customSlotsBlocked = domains.filter(
            d => d.domainType === 'custom' && d.status === 'phasing_out'
        ).length >= 2

        const isPremium = brokerage?.plan === 'premium'

        return useMemo(() => ({
            state: {
                domains,
                brokerage,
                isLoading: isLoadingDomains || isLoadingBrokerage,
                error: domainsError || brokerageError,
                slotWarning,
                subdomainSlotsBlocked,
                customSlotsBlocked,
                isPremium,
            },
            actions: {
                updateSubdomain,
                setPrimary,
                decommission,
                addCustom,
                removeCustom,
                setSlotWarning,
            },
        }), [
            domains, brokerage, isLoadingDomains, isLoadingBrokerage, domainsError, brokerageError,
            slotWarning, subdomainSlotsBlocked, customSlotsBlocked, isPremium,
            updateSubdomain, setPrimary, decommission, addCustom, removeCustom,
        ])
    }

    export default useDomainsAdminContext
    ```

            e.preventDefault()
            if (!isValid || disabled) return
            onSubmit(trimmed)
            setValue('')
        }

        return (
            <form onSubmit={handleSubmit} className="space-y-3 rounded-lg border border-border bg-card p-4">
                <Input
                    placeholder="app.acme.com"
                    value={value}
                    onChange={(e) => setValue(e.target.value)}
                    disabled={disabled}
                    aria-label="Custom domain"
                    autoFocus
                />
                <div className="flex items-center gap-2">
                    <Button type="submit" variant="default" disabled={!isValid || disabled}>
                        Add domain
                    </Button>
                    <Button type="button" variant="outline" onClick={onCancel} disabled={disabled}>
                        Discard
                    </Button>
                </div>
            </form>
        )
    }

    export default AddCustomDomainForm
