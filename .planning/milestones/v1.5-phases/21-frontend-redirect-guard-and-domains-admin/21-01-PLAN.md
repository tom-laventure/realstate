    ```ts
    // Public, unauthenticated lookup. Used ONLY by PrimaryDomainRedirect (DINT-01).
    // MUST use base `axios` — the project `api` instance attaches Authorization
    // and X-Brokerage-Id headers (see frontend/src/assets/axios/axios.ts), which
    // are inappropriate for this public endpoint and fire before auth is ready.
    import axios from 'axios'

    export interface LookupDomainResponse {
        domain: string
    }

    const lookupDomain = async (domain: string): Promise<LookupDomainResponse> => {
        const envBase = (import.meta.env.VITE_BE_ENDPOINT as string | undefined)?.trim()
        const baseUrl = envBase && envBase.length > 0 ? envBase : window.location.origin
        const response = await axios.get<LookupDomainResponse>(`${baseUrl}/v1/domains/lookup`, {
            params: { domain },
        })
        return response.data
    }

    export default lookupDomain
    ```

    Step B — Create `frontend/src/components/common/guards/PrimaryDomainRedirect.tsx` with EXACT content (D-01..D-04 from 21-CONTEXT.md):

    ```tsx
    import { useEffect, useState } from 'react'
    import { Outlet } from 'react-router-dom'
    import { isMainDomain } from '@/helpers/domain'
    import lookupDomain from '@/assets/api/domains/lookupDomain'

    /**
     * PrimaryDomainRedirect — layout route. On subdomain visits, looks up the
     * brokerage's primary domain and silently redirects there if it differs from
     * the current hostname. UX-only — not a security boundary.
     *
     * - D-01: skip entirely on the main domain.
     * - D-02: render null (blank screen — no spinner) while in flight.
     * - D-03: fail open on any error (network / 404 / 5xx).
     * - D-04: use window.location.replace so the phasing-out URL is not in history.
     */
    const PrimaryDomainRedirect = () => {
        const [ready, setReady] = useState(false)

        useEffect(() => {
            // D-01: main domain bypasses the guard entirely.
            if (isMainDomain()) {
                setReady(true)
                return
            }

            const hostname = window.location.hostname
            lookupDomain(hostname)
                .then((data) => {
                    if (data?.domain && data.domain !== hostname) {
                        // D-04: replace — phasing-out URL must NOT remain in history.
                        const primaryUrl = `https://${data.domain}${window.location.pathname}${window.location.search}`
                        window.location.replace(primaryUrl)
                        // Intentionally do NOT setReady — page is navigating away.
                    } else {
                        setReady(true)
                    }
                })
                .catch(() => {
                    // D-03: fail open on any error.
                    setReady(true)
                })
        }, [])

        // D-02: blank screen while resolving — no spinner, no flash.
        if (!ready) return null
        return <Outlet />
    }

    export default PrimaryDomainRedirect
    ```
