<!-- MainDomainOnly concern (from backend/app/controllers/concerns/main_domain_only.rb) -->

```ruby
module MainDomainOnly
  extend ActiveSupport::Concern

  included do
    before_action :require_main_domain!
  end

  private

  def require_main_domain!
    main_domain = ENV.fetch("APP_BASE_DOMAIN", "realstate.ca")
    origin_host = begin
      URI.parse(request.headers["Origin"].to_s).host&.downcase
    rescue URI::InvalidURIError
      nil
    end
    request_host = origin_host.presence || request.host.downcase
    dev_hosts = %w[localhost 127.0.0.1]
    allowed = request_host == main_domain || (Rails.env.development? && dev_hosts.include?(request_host))
    render json: {error: "Forbidden"}, status: :forbidden unless allowed
  end
end
```

**Step 1 — Create the controller.**

Create `backend/app/controllers/v1/account/admin_brokerages_controller.rb` with this exact content:

```ruby
module V1
  module Account
    class AdminBrokeragesController < ApplicationController
      include MainDomainOnly
      include CognitoAuth

      before_action :authenticate_cognito_user!

      # Skips brokerage membership check — no single brokerage context;
      # this endpoint is cross-brokerage by design. (D-07)
      def skip_brokerage_membership?
        true
      end

      # GET /v1/account/admin_brokerages
      # BE-01: returns all brokerages where Current.user holds admin or owner role.
      def index
        memberships = BrokerageMembership
          .without_deleted
          .includes(:brokerage)
          .where(user_id: Current.user.id, role: %w[admin owner])

        render json: {
          admin_brokerages: memberships.map { |m|
            {id: m.brokerage.id, name: m.brokerage.name, role: m.role}
          }
        }
      end
    end
  end
end
```

Key implementation rules:
- Include `MainDomainOnly` BEFORE `CognitoAuth` — `MainDomainOnly`'s `included do` registers `before_action :require_main_domain!`, which runs first (before authenticate_cognito_user!) per include order
- Do NOT include `BrokerageAuth` — no single brokerage context applies (D-06, anti-pattern in RESEARCH.md)
- Do NOT inherit from `V1::Admin::BaseController` — that base class includes BrokerageAuth and require_admin! (D-06)
- `BrokerageMembership.without_deleted.includes(:brokerage)` — explicit soft-delete scope (CLAUDE.md mandate) + N+1 prevention
- `render json: { admin_brokerages: [...] }` — inline render, no serializer (D-12)

```ruby
require "test_helper"

# Phase 15 — BE-01
# Verifies GET /v1/account/admin_brokerages:
#   - Returns all brokerages where user holds admin or owner role
#   - Returns empty list for users with no admin roles
#   - Returns 403 from brokerage subdomain
#   - Returns 401 for unauthenticated requests
class AdminBrokeragesTest < ActionDispatch::IntegrationTest
  MAIN_DOMAIN = ENV.fetch("APP_BASE_DOMAIN", "realstate.ca").freeze

  setup do
    @brokerage1 = create(:brokerage)
    @brokerage2 = create(:brokerage)

    @user = create(:user)
    create(:brokerage_membership, :admin, user: @user, brokerage: @brokerage1)
    create(:brokerage_membership, :owner, user: @user, brokerage: @brokerage2)
  end

  # BE-01: returns all admin/owner brokerages with correct fields
  test "GET /v1/account/admin_brokerages returns admin and owner brokerages with id, name, role" do
    get "/v1/account/admin_brokerages", headers: main_domain_headers_for(@user)
    assert_response :ok
    body = JSON.parse(response.body)
    assert_kind_of Array, body["admin_brokerages"]
    assert_equal 2, body["admin_brokerages"].length
    # Verify each entry has required fields
    body["admin_brokerages"].each do |entry|
      assert entry.key?("id"),   "entry missing 'id' key"
      assert entry.key?("name"), "entry missing 'name' key"
      assert entry.key?("role"), "entry missing 'role' key"
      assert_includes %w[admin owner], entry["role"]
    end
  end

  # BE-01: agent-only user gets an empty list (not an error)
  test "GET /v1/account/admin_brokerages returns empty list when user has no admin or owner roles" do
    agent = create(:user)
    brokerage3 = create(:brokerage)
    create(:brokerage_membership, :agent, user: agent, brokerage: brokerage3)

    get "/v1/account/admin_brokerages", headers: main_domain_headers_for(agent)
    assert_response :ok
    body = JSON.parse(response.body)
    assert_equal [], body["admin_brokerages"]
  end

  # BE-01 (access control): endpoint is restricted to main domain
  test "GET /v1/account/admin_brokerages from brokerage subdomain returns 403" do
    get "/v1/account/admin_brokerages",
      headers: main_domain_headers_for(@user).merge("Origin" => "https://acme.#{MAIN_DOMAIN}")
    assert_response :forbidden
  end

  # BE-01 (auth): unauthenticated request returns 401
  test "GET /v1/account/admin_brokerages without auth token returns 401" do
    get "/v1/account/admin_brokerages",
      headers: {"Origin" => "https://#{MAIN_DOMAIN}"}
    assert_response :unauthorized
  end

  private

  # Mints token without Current.brokerage — no single brokerage context for this endpoint.
  # Uses Origin header (not Host) so MainDomainOnly and BrokerageAuth both see the main domain.
  def main_domain_headers_for(user)
    Current.user = user
    token = TokenService.issue(user)
    {
      "Authorization" => "Bearer #{token}",
      "Origin" => "https://#{MAIN_DOMAIN}"
    }
  ensure
    Current.reset
  end
end
```

Notes on implementation:
- `main_domain_headers_for` does NOT set `Current.brokerage` — no single brokerage context for this cross-brokerage endpoint
- Uses `Origin` header (not `Host`) so `MainDomainOnly#require_main_domain!` passes using the same `extract_domain_from_request` logic
- Subdomain test overrides the `Origin` header with a subdomain value — MainDomainOnly rejects with 403
- Unauthenticated test omits Authorization header — CognitoAuth rejects with 401

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → MainDomainOnly | Origin/Host header determines if request passes the domain guard; attacker-controlled |
| client → CognitoAuth | Bearer JWT in Authorization header; verified cryptographically by TokenService |
| controller → BrokerageMembership | user_id comes from Current.user (set by CognitoAuth after JWT verification); role filter is server-side |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-15-06 | Spoofing | `GET /v1/account/admin_brokerages` unauthenticated | mitigate | `before_action :authenticate_cognito_user!` rejects with 401 before index runs |
| T-15-07 | EoP | Non-admin user calls endpoint and receives empty list, then tries to infer brokerage existence | accept | Empty list for agent-only users is intentional (BE-01); reveals no sensitive data — response is scoped to the authenticated user's own memberships only |
| T-15-08 | EoP | User manipulates response to claim admin role they don't have | accept | Response is read-only; membership data is queried from DB with server-side user_id from verified JWT — client cannot inject role values |
| T-15-09 | Tampering | Subdomain user sends X-Brokerage-Id header to admin_brokerages endpoint | accept | BrokerageAuth is not included — no header is read. MainDomainOnly rejects subdomain requests with 403 before authenticate_cognito_user! runs |
| T-15-10 | Information Disclosure | Soft-deleted brokerage memberships returned | mitigate | `BrokerageMembership.without_deleted` explicitly excludes soft-deleted records (CLAUDE.md mandate) |
| T-15-11 | DoS / N+1 | Mapping over memberships triggers per-record brokerage queries | mitigate | `.includes(:brokerage)` preloads all brokerage records in a single query — no per-iteration DB hit |
</threat_model>
