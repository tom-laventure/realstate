```ruby
module BrokerageAuth
  extend ActiveSupport::Concern

  included do
    before_action :resolve_brokerage
  end

  private

  def resolve_brokerage
    domain = extract_domain_from_request

    logger.debug("Resolving brokerage for domain: #{domain}")

    domain_record = BrokerageDomain
      .includes(:brokerage)
      .find_by(domain: domain)

    raise ActiveRecord::RecordNotFound, "Brokerage not found for domain: #{domain}" unless domain_record

    Current.brokerage = domain_record.brokerage
  end

  def extract_domain_from_request
    origin_host = begin
      URI.parse(request.headers["Origin"].to_s).host&.downcase
    rescue URI::InvalidURIError
      nil
    end

    origin_host.presence || request.host.downcase
  end
end
```

```ruby
module BrokerageAuth
  extend ActiveSupport::Concern

  included do
    before_action :resolve_brokerage
  end

  private

  def resolve_brokerage
    if main_domain_request?
      header_val = request.headers["X-Brokerage-Id"]
      unless header_val.present?
        render json: {error: "X-Brokerage-Id header required"}, status: :bad_request
        return
      end
      brokerage = Brokerage.without_deleted.find_by(id: header_val.to_i)
      raise ActiveRecord::RecordNotFound, "Brokerage not found for id: #{header_val}" unless brokerage
      Current.brokerage = brokerage
    else
      domain = extract_domain_from_request
      logger.debug("Resolving brokerage for domain: #{domain}")
      domain_record = BrokerageDomain
        .includes(:brokerage)
        .find_by(domain: domain)
      raise ActiveRecord::RecordNotFound, "Brokerage not found for domain: #{domain}" unless domain_record
      Current.brokerage = domain_record.brokerage
    end
  end

  def main_domain_request?
    main_domain = ENV.fetch("APP_BASE_DOMAIN", "realstate.ca")
    request_host = extract_domain_from_request
    dev_hosts = %w[localhost 127.0.0.1]
    request_host == main_domain || (Rails.env.development? && dev_hosts.include?(request_host))
  end

  def extract_domain_from_request
    origin_host = begin
      URI.parse(request.headers["Origin"].to_s).host&.downcase
    rescue URI::InvalidURIError
      nil
    end
    origin_host.presence || request.host.downcase
  end
end
```

- `return` immediately after `render` to prevent AbstractController::DoubleRenderError (per D-04 and pitfall #2 in RESEARCH.md)
- Use `Brokerage.without_deleted.find_by(id: header_val.to_i)` — explicit `.without_deleted` per CLAUDE.md mandate (pitfall #4)

```ruby
# Phase 15 — BE-02 / BE-03
# Verifies BrokerageAuth#resolve_brokerage main-domain branch:
#   BE-02: X-Brokerage-Id header on main-domain request resolves Current.brokerage correctly
#   BE-03: Missing X-Brokerage-Id header on main-domain request returns HTTP 400
class BrokerageAuthMainDomainTest < ActionDispatch::IntegrationTest
  MAIN_DOMAIN = ENV.fetch("APP_BASE_DOMAIN", "realstate.ca").freeze

  setup do
    @brokerage = create(:brokerage)
    create(:brokerage_domain, brokerage: @brokerage, domain: "acme.#{MAIN_DOMAIN}")
    @admin = create(:user)
    create(:brokerage_membership, :admin, user: @admin, brokerage: @brokerage)
  end

  # BE-02: X-Brokerage-Id header present resolves brokerage and returns 200
  test "GET /v1/admin/users from main domain with X-Brokerage-Id resolves brokerage and returns 200" do
    headers = main_domain_headers_for(@admin, @brokerage.id)
    get "/v1/admin/users", headers: headers
    assert_response :ok
  end

  # BE-03: X-Brokerage-Id header absent returns 400
  test "GET /v1/admin/users from main domain without X-Brokerage-Id returns 400" do
    headers = main_domain_headers_for(@admin, nil)
    get "/v1/admin/users", headers: headers
    assert_response :bad_request
    body = JSON.parse(response.body)
    assert_equal "X-Brokerage-Id header required", body["error"]
  end

  # BE-02 (isolation): subdomain request is unaffected by main-domain branch
  test "GET /v1/admin/users from subdomain with Host header still resolves via BrokerageDomain" do
    headers = auth_headers_for(@admin, @brokerage)
    get "/v1/admin/users", headers: headers
    assert_response :ok
  end

  # BE-03 (security): invalid brokerage ID returns 404 (not found, not 500)
  test "GET /v1/admin/users from main domain with non-existent X-Brokerage-Id returns 404" do
    headers = main_domain_headers_for(@admin, 99999)
    get "/v1/admin/users", headers: headers
    assert_response :not_found
  end

  private

  # Mint token for main-domain requests. Do NOT use auth_headers_for — it sets
  # Current.brokerage and uses Host header, which triggers the subdomain path.
  def main_domain_headers_for(user, brokerage_id)
    Current.user = user
    token = TokenService.issue(user)
    headers = {
      "Authorization" => "Bearer #{token}",
      "Origin" => "https://#{MAIN_DOMAIN}"
    }
    headers["X-Brokerage-Id"] = brokerage_id.to_s if brokerage_id.present?
    headers
  ensure
    Current.reset
  end
end
```

Notes:
- `main_domain_headers_for` does NOT set `Current.brokerage` before `TokenService.issue` — main-domain context has no single brokerage
- Sets `Origin: https://#{MAIN_DOMAIN}` (not `Host`) so `extract_domain_from_request` returns the main domain
- `auth_headers_for` (from AuthHelper) is used only in the subdomain isolation test — it uses `Host` header and sets `Current.brokerage`, which is correct for the subdomain path
