```typescript
import api from '@/assets/axios/axios'
import { EstateCoordinate } from '@/assets/types/estateCoordinateType'

interface FetchGroupEstateCoordinatesPayload {
    groupId: string
    search?: string
    showFavoritesOnly?: boolean
    source?: string
    cities?: string[]
    propertyTypes?: string[]
    minPrice?: string
    maxPrice?: string
    minPricePerSqft?: string
    maxPricePerSqft?: string
    maxMaintenanceFee?: string
    maxDaysListed?: string
    minBedrooms?: string
    minBathrooms?: string
    minSqft?: string
    maxAgeYears?: string
    minUserRating?: string
    minGroupRating?: string
    hasComments?: boolean
    hasSharedDocuments?: boolean
    pendingReview?: boolean
    showRejected?: boolean
    maxDaysAddedToGroup?: string
    sortBy?: string
}

interface FetchGroupEstateCoordinatesResponse {
    coordinates: EstateCoordinate[]
}

const fetchGroupEstateCoordinates = async ({
    groupId,
    search,
    showFavoritesOnly,
    source,
    cities,
    propertyTypes,
    minPrice,
    maxPrice,
    minPricePerSqft,
    maxPricePerSqft,
    maxMaintenanceFee,
    maxDaysListed,
    minBedrooms,
    minBathrooms,
    minSqft,
    maxAgeYears,
    minUserRating,
    minGroupRating,
    hasComments,
    hasSharedDocuments,
    pendingReview,
    showRejected,
    maxDaysAddedToGroup,
    sortBy,
}: FetchGroupEstateCoordinatesPayload): Promise<FetchGroupEstateCoordinatesResponse> => {
    const response = await api.get(`/v1/dashboard/groups/${groupId}/estates/coordinates`, {
        params: {
            ...(search && { search }),
            ...(showFavoritesOnly && { favorites_only: showFavoritesOnly }),
            ...(source && { source }),
            ...(cities && cities.length > 0 && { 'cities[]': cities }),
            ...(propertyTypes && propertyTypes.length > 0 && { 'property_types[]': propertyTypes }),
            ...(minPrice && { min_price: minPrice }),
            ...(maxPrice && { max_price: maxPrice }),
            ...(minPricePerSqft && { min_price_per_sqft: minPricePerSqft }),
            ...(maxPricePerSqft && { max_price_per_sqft: maxPricePerSqft }),
            ...(maxMaintenanceFee && { max_maintenance_fee: maxMaintenanceFee }),
            ...(maxDaysListed && { max_days_listed: maxDaysListed }),
            ...(minBedrooms && { min_bedrooms: minBedrooms }),
            ...(minBathrooms && { min_bathrooms: minBathrooms }),
            ...(minSqft && { min_sqft: minSqft }),
            ...(maxAgeYears && { max_age_years: maxAgeYears }),
            ...(minUserRating && { min_user_rating: minUserRating }),
            ...(minGroupRating && { min_group_rating: minGroupRating }),
            ...(hasComments && { has_comments: hasComments }),
            ...(hasSharedDocuments && { has_shared_documents: hasSharedDocuments }),
            ...(pendingReview && { pending_review: pendingReview }),
            ...(showRejected && { show_rejected: showRejected }),
            ...(maxDaysAddedToGroup && { max_days_added_to_group: maxDaysAddedToGroup }),
            ...(sortBy && { sort_by: sortBy }),
        },
    })
    return response.data
}

export type { FetchGroupEstateCoordinatesPayload, FetchGroupEstateCoordinatesResponse }
export default fetchGroupEstateCoordinates
```
