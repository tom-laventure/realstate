**Step 1 — Create frontend/src/components/common/guards/DomainGuard.tsx:**

Single file with two named exports. Follow exact ProtectedRoute Outlet pattern.

```typescript
import { Navigate, Outlet } from 'react-router-dom'
import { isMainDomain } from '@/helpers/domain'
import NotFound from '@/containers/errorPage/NotFound'

/**
 * SubdomainOnly — blocks access on the main domain.
 * On main domain: renders NotFound. On brokerage subdomain: passes through via Outlet.
 *
 * UX enforcement only — not a security boundary.
 * Phase 7 adds backend 403 enforcement for onboarding endpoints.
 */
export const SubdomainOnly = () => {
  if (isMainDomain()) return <NotFound />
  return <Outlet />
}

/**
 * MainDomainOnly — blocks access on brokerage subdomains.
 * On brokerage subdomain: redirects to / (replace, no history entry). On main domain: passes through via Outlet.
 *
 * UX enforcement only — not a security boundary.
 * Phase 7 adds backend 403 enforcement for onboarding endpoints.
 */
export const MainDomainOnly = () => {
  if (!isMainDomain()) return <Navigate to="/" replace />
  return <Outlet />
}
```

No default export — named exports only (SubdomainOnly, MainDomainOnly).
